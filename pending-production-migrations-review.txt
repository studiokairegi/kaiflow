-- Keep this migration independent of shared-domain helpers introduced by later migrations.
create or replace function public._intel_indie_domain(p_value text)
returns text language sql immutable set search_path = pg_catalog as $$
  select regexp_replace(
    split_part(split_part(regexp_replace(lower(btrim(coalesce(p_value, ''))), '^https?://', ''), '/', 1), ':', 1),
    '^www\.', '')
$$;

alter table public.intel_prospects

  add column if not exists project_source_url text not null default '',
  add column if not exists project_announcement_date date,

  add column if not exists project_expected_release_date date,

  add column if not exists project_latest_activity_date date,
  add column if not exists project_marketing_milestones jsonb not null default '[]'::jsonb;

create or replace function public._intel_try_project_date(p_value text)
returns date language plpgsql immutable set search_path = pg_catalog as $$
begin
  if p_value is null or btrim(p_value) = '' then return null; end if;
  if btrim(p_value) ~ '^[0-9]{4}$' then
    begin return make_date(btrim(p_value)::integer, 7, 1); exception when others then return null; end;
  end if;
  begin return btrim(p_value)::date; exception when others then null; end;
  begin return left(btrim(p_value), 10)::date; exception when others then null; end;
  begin return to_date(btrim(p_value), 'YYYY-MM'); exception when others then null; end;
  begin return to_date(btrim(p_value), 'Mon YYYY'); exception when others then null; end;
  begin return to_date(btrim(p_value), 'Month YYYY'); exception when others then null; end;
  begin return to_date(btrim(p_value), 'Mon DD, YYYY'); exception when others then null; end;
  return null;
end $$;

create or replace function public._intel_is_protected_indie_entity(p_developer text, p_project text)
returns boolean language sql immutable set search_path = pg_catalog as $$
  select regexp_replace(lower(coalesce(p_developer, '')), '[^a-z0-9]', '', 'g') = any(array['brainjargames','brainjargamesllc','brainjargamesinc','silverlineinteractive','silverlineinteractiveinc','silverlineinteractivellc','thermitegames','thermitegamesllc','thermitegamesinc','lieofcaelumspirit'])
      or regexp_replace(lower(coalesce(p_project, '')), '[^a-z0-9]', '', 'g') = any(array['brainjargames','brainjargamesllc','brainjargamesinc','silverlineinteractive','silverlineinteractiveinc','silverlineinteractivellc','thermitegames','thermitegamesllc','thermitegamesinc','lieofcaelumspirit']);
$$;

create or replace function public._intel_sync_project_facts()
returns trigger language plpgsql security definer set search_path = pg_catalog, public as $$
declare v jsonb := coalesce(new.facts -> 'verified', '{}'::jsonb); v_coming boolean; v_project_url text;
begin
  if new.type <> 'indie_game_developer' then return new; end if;
  new.project_title := coalesce(nullif(v ->> 'project_title', ''), nullif(v ->> 'game', ''), new.project_title);
  new.project_source_url := coalesce(nullif(v ->> 'project_url', ''), nullif(v ->> 'game_url', ''),
    case when coalesce(v ->> 'steam_appid', '') ~ '^[0-9]+$' then 'https://store.steampowered.com/app/' || (v ->> 'steam_appid') || '/' else null end, new.project_source_url);
  if btrim(new.project_source_url) <> '' and new.project_source_url !~* '^https?://' then
    new.project_source_url := 'https://' || btrim(new.project_source_url);
  end if;
  if (btrim(new.project_source_url) = '' or new.project_source_url !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+')
     and coalesce(v ->> 'steam_appid', '') ~ '^[0-9]+$' then
    new.project_source_url := 'https://store.steampowered.com/app/' || (v ->> 'steam_appid') || '/';
  end if;
  if btrim(new.project_source_url) = '' or new.project_source_url !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' then
    select e.url into v_project_url from public.intel_evidence e
     where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
       and (e.fact_type in ('game_store_page','steam_project','itch_project','kickstarter_project')
         or (e.fact_type in ('project_identity','project_description') and (public._intel_indie_domain(e.url) = public._intel_indie_domain(new.website)
           or public._intel_indie_domain(e.url) = public._intel_indie_domain(new.project_source_url))))
     order by e.fetched_at desc limit 1;
    new.project_source_url := coalesce(v_project_url, new.project_source_url);
  end if;
  new.project_announcement_date := coalesce(public._intel_try_project_date(v ->> 'project_announcement_date'), public._intel_try_project_date(v ->> 'announcement_date'), new.project_announcement_date);
  new.project_campaign_date := coalesce(public._intel_try_project_date(v ->> 'project_campaign_date'), public._intel_try_project_date(v ->> 'campaign_date'), new.project_campaign_date);
  v_coming := (v ->> 'coming_soon') = 'true';
  new.project_expected_release_date := coalesce(public._intel_try_project_date(v ->> 'project_expected_release_date'), public._intel_try_project_date(v ->> 'expected_release_date'), new.project_expected_release_date,
    case when v_coming then public._intel_try_project_date(v ->> 'release_date') else null end);
  new.project_actual_release_date := coalesce(public._intel_try_project_date(v ->> 'project_actual_release_date'), public._intel_try_project_date(v ->> 'actual_release_date'), new.project_actual_release_date,
    case when (v ->> 'coming_soon') = 'false' then public._intel_try_project_date(v ->> 'release_date') else null end);
  new.project_latest_activity_date := coalesce(public._intel_try_project_date(v ->> 'project_latest_activity_date'), public._intel_try_project_date(v ->> 'latest_activity_date'), new.project_latest_activity_date);
  if jsonb_typeof(v -> 'project_marketing_milestones') = 'array' then
    new.project_marketing_milestones := v -> 'project_marketing_milestones';
  elsif jsonb_typeof(v -> 'marketing_milestones') = 'array' then
    new.project_marketing_milestones := v -> 'marketing_milestones';
  end if;
  if exists (select 1 from public.intel_suppressions x where x.organization_id = new.organization_id
      and ((x.kind = 'email' and x.value = public.intel_norm_email(new.contact_email) and public.intel_norm_email(new.contact_email) <> '')
        or (x.kind = 'domain' and (x.value = public._intel_indie_domain(new.domain)
          or public.intel_norm_email(new.contact_email) like '%@' || x.value)))) then
    new.status := 'suppressed';
    new.review_reason := 'Existing email/domain suppression conflict';
  end if;
  if public._intel_is_protected_indie_entity(new.name, coalesce(nullif(new.project_title, ''), v ->> 'project_title', v ->> 'game')) then
    new.status := 'rejected';
    new.review_reason := 'Protected Studio Kairegi account';
  end if;
  return new;
end $$;

create or replace function public._intel_enforce_indie_qualification_status()
returns trigger language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_gates jsonb;
  v_gate text;
  v_required text[] := array['project_identity','active_or_marketing_opportunity','anime_visual_identity','verified_usable_email','suppression_clear','protected_account_clear','sufficient_evidence'];
  v_email text;
  v_project_url text;
begin
  if new.type <> 'indie_game_developer' or new.status <> 'qualified' then return new; end if;
  if tg_op = 'UPDATE' and old.status = 'suppressed' then
    raise exception 'suppressed indie prospect cannot be requalified';
  end if;
  v_gates := new.facts #> '{estimates,qualification_v2,mandatory_gates}';
  v_email := public.intel_norm_email(new.contact_email);
  if new.score is null or new.score < 75 then raise exception 'indie V2 prospect cannot be qualified below 75 points'; end if;
  if coalesce(new.facts #>> '{estimates,qualification_v2,model}', '') <> 'indie-project-v2' then
    raise exception 'indie V2 qualification model version is required';
  end if;
  if v_gates is null then raise exception 'indie V2 mandatory qualification gates are required'; end if;
  foreach v_gate in array v_required loop
    if coalesce(v_gates -> v_gate ->> 'passed', 'false') <> 'true' then
      raise exception 'indie V2 mandatory gate failed: %', v_gate;
    end if;
  end loop;
  if coalesce(new.facts #>> '{verified,contactability,status}', '') <> 'pass'
     or coalesce(new.facts #>> '{verified,bounced}', 'false') = 'true'
     or coalesce(new.facts #>> '{verified,complained}', 'false') = 'true'
     or coalesce(new.facts #>> '{verified,complaint}', 'false') = 'true'
     or coalesce(new.facts #>> '{verified,unsubscribed}', 'false') = 'true'
     or coalesce(new.facts #>> '{verified,do_not_contact}', 'false') = 'true'
     or coalesce(new.facts #>> '{verified,suppressed}', 'false') = 'true'
     or v_email = ''
     or public.intel_norm_email(new.facts #>> '{verified,contactability,verified_email}') <> v_email
     or coalesce(new.facts #>> '{verified,contactability,evidence,source_url}', '') !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' then
    raise exception 'indie V2 verified contactability must pass for the exact sourced email';
  end if;
  if btrim(new.project_source_url) <> '' and new.project_source_url !~* '^https?://' then
    new.project_source_url := 'https://' || btrim(new.project_source_url);
  end if;
  if (btrim(new.project_source_url) = '' or new.project_source_url !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+')
     and coalesce(new.facts #>> '{verified,steam_appid}', '') ~ '^[0-9]+$' then
    new.project_source_url := 'https://store.steampowered.com/app/' || (new.facts #>> '{verified,steam_appid}') || '/';
  end if;
  if btrim(new.project_source_url) = '' or new.project_source_url !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' then
    select e.url into v_project_url from public.intel_evidence e
     where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
       and (e.fact_type in ('game_store_page','steam_project','itch_project','kickstarter_project')
         or (e.fact_type in ('project_identity','project_description') and (public._intel_indie_domain(e.url) = public._intel_indie_domain(new.website)
           or public._intel_indie_domain(e.url) = public._intel_indie_domain(new.project_source_url))))
     order by e.fetched_at desc limit 1;
    new.project_source_url := coalesce(v_project_url, new.project_source_url);
  end if;
  if btrim(new.project_title) = '' or regexp_replace(lower(btrim(new.project_title)), '[^a-z0-9]', '', 'g') in ('unknown','unknowngame','tbd','na','notavailable','unannounced')
     or btrim(new.name) = '' or regexp_replace(lower(btrim(new.name)), '[^a-z0-9]', '', 'g') in ('unknown','unknowndeveloper','tbd','na','notavailable','anonymous')
     or new.project_source_url !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' then
    raise exception 'indie V2 project identity must include an identifiable developer, title and HTTP(S) project source';
  end if;
  if not exists (select 1 from public.intel_evidence e where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
      and e.fact_type in ('anime_visual_evidence','anime_visual_screenshot','anime_visual_artwork','anime_visual_trailer','anime_character_art','official_anime_visuals_verified'))
     and not exists (select 1 from public.intel_evidence e where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
       and e.fact_type in ('anime_style_description','official_anime_style_description','official_visual_style')
       and (public._intel_indie_domain(e.url) = public._intel_indie_domain(new.website) or public._intel_indie_domain(e.url) = public._intel_indie_domain(new.project_source_url))
       and coalesce(e.snippet, '') ~* '(anime|manga).{0,60}(style|inspired|art|visuals?|aesthetics?|character design|illustration)|(style|art|visuals?|aesthetics?).{0,60}(anime|manga)')) then
    raise exception 'indie V2 requires verified anime-compatible visual or explicit official style evidence';
  end if;
  if not (
      (new.project_actual_release_date is null and new.project_latest_activity_date is not null and new.project_latest_activity_date between current_date - 365 and current_date and exists (
        select 1 from public.intel_evidence e where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
          and e.fact_type in ('devlog','development_log','itch_devlog','steam_devlog','kickstarter_development_update','developer_progress_update','development_update','project_update','demo_update','playtest_update','build_update','patch_update')))
      or (new.project_actual_release_date is null and new.project_expected_release_date is not null and new.project_expected_release_date between current_date and current_date + 365)
      or (new.project_actual_release_date is null and coalesce(new.facts #>> '{verified,coming_soon}', 'false') = 'true'
          and (new.project_expected_release_date is null or new.project_expected_release_date between current_date and current_date + 365))
      or (new.project_actual_release_date is null and new.project_announcement_date between current_date - 240 and current_date - 120
          and (new.project_expected_release_date is null or new.project_expected_release_date >= current_date))
      or (new.project_campaign_date is not null and new.project_campaign_date between current_date - 30 and current_date + 365)
      or exists (select 1 from jsonb_array_elements(case when jsonb_typeof(new.project_marketing_milestones) = 'array' then new.project_marketing_milestones else '[]'::jsonb end) as milestone(value)
        where milestone.value ->> 'verified' = 'true'
          and coalesce(milestone.value ->> 'source_url', milestone.value ->> 'url', '') <> ''
          and public._intel_try_project_date(coalesce(milestone.value ->> 'date', milestone.value ->> 'start_date', milestone.value ->> 'campaign_date')) between current_date - 30 and current_date + 365
          and exists (select 1 from public.intel_evidence e where e.prospect_id = new.id and e.verified
            and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
            and e.fact_type in ('dlc_announcement','major_update_announcement','relaunch_announcement','console_release_announcement','marketing_campaign','marketing_opportunity')
            and rtrim(e.url, '/') = rtrim(coalesce(milestone.value ->> 'source_url', milestone.value ->> 'url'), '/')))
      or (new.project_actual_release_date is null and exists (select 1 from public.intel_evidence e where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> ''
          and e.fact_type in ('dlc_announcement','major_update_announcement','relaunch_announcement','console_release_announcement','marketing_campaign','marketing_opportunity')
          and e.snippet ~* '(upcoming|coming soon|launch(es|ed)?|pre.?orders?|wishlists?|demos?|early access|campaigns?|DLC|major updates?|console releases?|relaunch(es|ed)?|expansions?)'))
    ) then
    raise exception 'indie V2 requires dated activity or a credible marketing opportunity';
  end if;
  if (select count(distinct e.fact_type) from public.intel_evidence e where e.prospect_id = new.id and e.verified and e.url ~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' and coalesce(e.snippet, '') <> '') < 2 then
    raise exception 'indie V2 requires at least two distinct verified evidence types';
  end if;
  if public._intel_is_protected_indie_entity(new.name, coalesce(nullif(new.project_title, ''), new.facts #>> '{verified,project_title}', new.facts #>> '{verified,game}')) then
    raise exception 'protected Studio Kairegi account cannot be qualified';
  end if;
  if exists (select 1 from public.intel_suppressions x where x.organization_id = new.organization_id and
    ((x.kind = 'email' and x.value = v_email and v_email <> '') or (x.kind = 'domain' and (x.value = public._intel_indie_domain(new.domain) or v_email like '%@' || x.value)))) then
    raise exception 'indie V2 prospect has an active email or domain suppression';
  end if;
  return new;
end $$;

create or replace function public._intel_block_protected_indie_outbound()
returns trigger language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_prospect public.intel_prospects;
  v_email text;
  v_lead_name text;
  v_gates jsonb;
  v_send_transition boolean := false;
begin
  if new.direction <> 'outbound' then return new; end if;
  if tg_op = 'INSERT' then
    v_send_transition := new.state in ('reserved','sending');
  else
    v_send_transition := new.state in ('reserved','sending') and old.state is distinct from new.state;
  end if;
  if not v_send_transition then return new; end if;

  if new.prospect_id is not null then
    select * into v_prospect from public.intel_prospects
     where id = new.prospect_id and organization_id = new.organization_id;
  elsif new.lead_id is not null then
    select * into v_prospect from public.intel_prospects
     where lead_id = new.lead_id and organization_id = new.organization_id and type = 'indie_game_developer'
     order by created_at desc limit 1;
  end if;
  if v_prospect.id is not null and new.campaign_type = 'indie_game_developer' and v_prospect.type <> 'indie_game_developer' then
    raise exception 'indie campaign message must reference an indie prospect';
  end if;
  if v_prospect.id is null and new.campaign_type = 'indie_game_developer' then
    if new.kind = 'reply' and new.lead_id is null then
      raise exception 'indie reply must be linked to an existing prospect or lead';
    end if;
    if new.lead_id is not null then
      select l.company_name into v_lead_name from public.leads l where l.id = new.lead_id;
    end if;
    if public._intel_is_protected_indie_entity(coalesce(v_lead_name, ''), '') then
      raise exception 'outbound contact to protected Studio Kairegi account is blocked';
    end if;
    v_email := public.intel_norm_email(new.to_email);
    if exists (select 1 from public.intel_suppressions x where x.organization_id = new.organization_id
      and ((x.kind = 'email' and x.value = v_email and v_email <> '')
        or (x.kind = 'domain' and (x.value = public._intel_indie_domain(split_part(v_email, '@', 2)) or v_email like '%@' || x.value)))) then
      raise exception 'outbound contact to suppressed indie developer is blocked';
    end if;
    if new.kind <> 'reply' then
      raise exception 'outbound indie prospecting requires a linked and qualified project';
    end if;
  end if;

  if v_prospect.id is not null and v_prospect.type = 'indie_game_developer' then
    v_email := public.intel_norm_email(v_prospect.contact_email);
    if public._intel_is_protected_indie_entity(v_prospect.name, coalesce(nullif(v_prospect.project_title, ''), v_prospect.facts #>> '{verified,project_title}', v_prospect.facts #>> '{verified,game}')) then
      raise exception 'outbound contact to protected Studio Kairegi account is blocked';
    end if;
    if v_prospect.status = 'suppressed'
       or coalesce(v_prospect.facts #>> '{verified,bounced}', 'false') = 'true'
       or coalesce(v_prospect.facts #>> '{verified,complained}', 'false') = 'true'
       or coalesce(v_prospect.facts #>> '{verified,complaint}', 'false') = 'true'
       or coalesce(v_prospect.facts #>> '{verified,unsubscribed}', 'false') = 'true'
       or coalesce(v_prospect.facts #>> '{verified,do_not_contact}', 'false') = 'true'
       or coalesce(v_prospect.facts #>> '{verified,suppressed}', 'false') = 'true'
       or exists (select 1 from public.intel_suppressions x where x.organization_id = v_prospect.organization_id
         and ((x.kind = 'email' and x.value = v_email and v_email <> '')
           or (x.kind = 'domain' and (x.value = public._intel_indie_domain(v_prospect.domain) or v_email like '%@' || x.value)))) then
      raise exception 'outbound contact to suppressed indie developer is blocked';
    end if;
    if coalesce(v_prospect.facts #>> '{verified,contactability,status}', '') <> 'pass'
       or v_email = ''
       or public.intel_norm_email(v_prospect.facts #>> '{verified,contactability,verified_email}') <> v_email
       or public.intel_norm_email(new.to_email) <> v_email
       or coalesce(v_prospect.facts #>> '{verified,contactability,evidence,source_url}', '') !~* '^https?://[^/[:space:]]*\.[^/[:space:]]+' then
      raise exception 'outbound indie contact requires exact-address verified contactability';
    end if;
    if new.kind <> 'reply' then
      if v_prospect.status not in ('qualified','lead_linked','contacted') then
        raise exception 'cold indie follow-up is blocked for a prospect outside an active qualified state';
      end if;
      v_gates := v_prospect.facts #> '{estimates,qualification_v2}';
      if coalesce(v_gates ->> 'model', '') <> 'indie-project-v2'
         or coalesce(v_prospect.score, 0) < 75
         or (select count(*) from jsonb_each(coalesce(v_gates -> 'mandatory_gates', '{}'::jsonb))) <> 7
         or exists (select 1 from jsonb_each(coalesce(v_gates -> 'mandatory_gates', '{}'::jsonb)) g where coalesce(g.value ->> 'passed', 'false') <> 'true') then
        raise exception 'outbound indie prospecting requires V2 score and all mandatory gates';
      end if;
    end if;
  end if;
  return new;
end $$;

-- Reconcile old approvals against current safety facts. CRM leads, messages and
-- conversation history are never deleted. Legacy qualified indie prospects are
-- moved to Needs Review until they are rescored under the V2 model.
update public.intel_prospects p set status = 'rejected', review_reason = 'Protected Studio Kairegi account', updated_at = now()
where p.type = 'indie_game_developer' and p.status <> 'rejected'
  and public._intel_is_protected_indie_entity(p.name, coalesce(nullif(p.project_title, ''), p.facts #>> '{verified,project_title}', p.facts #>> '{verified,game}'));

update public.intel_prospects p set status = 'suppressed', review_reason = 'Existing email/domain suppression conflict', updated_at = now()
where p.type = 'indie_game_developer' and p.status not in ('rejected','closed','suppressed')
  and exists (select 1 from public.intel_suppressions x where x.organization_id = p.organization_id
    and ((x.kind = 'email' and x.value = public.intel_norm_email(p.contact_email) and public.intel_norm_email(p.contact_email) <> '')
      or (x.kind = 'domain' and (x.value = public._intel_indie_domain(p.domain) or public.intel_norm_email(p.contact_email) like '%@' || x.value))));

update public.intel_prospects p set status = 'needs_review',
  review_reason = 'Legacy indie qualification must be re-evaluated under indie-project-v2', updated_at = now()
where p.type = 'indie_game_developer' and p.status = 'qualified';

-- This update fires the ordered project-fact sync trigger for existing rows.
update public.intel_prospects p set facts = p.facts
where p.type = 'indie_game_developer';

create or replace function public.intel_agent_set_score(
  p_prospect_id uuid, p_score int, p_breakdown jsonb, p_decision text,
  p_reason text default '', p_estimates jsonb default '{}'::jsonb
) returns void
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  p public.intel_prospects;
  v_gates jsonb;
  v_gate_names text[] := array['project_identity','active_or_marketing_opportunity','anime_visual_identity','verified_usable_email','suppression_clear','protected_account_clear','sufficient_evidence'];
  v_gates_ok boolean := true;
  v_gate text;
  v_email text;
begin
  if p_decision not in ('qualified','needs_review','rejected') then raise exception 'bad decision'; end if;
  select * into p from public.intel_prospects where id = p_prospect_id and organization_id = v_org for update;
  if not found then raise exception 'prospect not found'; end if;
  if p.type = 'indie_game_developer' and p_decision = 'qualified' then
    v_gates := p_estimates #> '{qualification_v2,mandatory_gates}';
    if v_gates is null then raise exception 'indie V2 mandatory qualification gates are required'; end if;
    foreach v_gate in array v_gate_names loop
      if coalesce(v_gates -> v_gate ->> 'passed', 'false') <> 'true' then v_gates_ok := false; end if;
    end loop;
    v_email := public.intel_norm_email(p.contact_email);
    if not v_gates_ok then raise exception 'indie V2 mandatory qualification gates have not all passed'; end if;
    if coalesce(p_estimates #>> '{contactability,status}', '') <> 'pass'
       or coalesce(p.facts #>> '{verified,bounced}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,complained}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,complaint}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,unsubscribed}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,do_not_contact}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,suppressed}', 'false') = 'true'
       or v_email = ''
       or public.intel_norm_email(p_estimates #>> '{contactability,verified_email}') <> v_email then
      raise exception 'indie V2 exact-address contactability must pass';
    end if;
    if exists (select 1 from public.intel_suppressions s where s.organization_id = v_org and
      ((s.kind = 'email' and s.value = v_email and v_email <> '') or (s.kind = 'domain' and (s.value = public._intel_indie_domain(p.domain) or (v_email like '%@' || s.value))))) then
      raise exception 'indie V2 prospect has an active email or domain suppression';
    end if;
    if p_score < 75 then raise exception 'indie V2 qualification requires at least 75 points'; end if;
    if coalesce(p_estimates #>> '{qualification_v2,model}', '') <> 'indie-project-v2' then
      raise exception 'indie V2 qualification model version is required';
    end if;
  end if;
  update public.intel_prospects
     set score = least(100, greatest(0, p_score)), score_breakdown = coalesce(p_breakdown, '{}'::jsonb),
         status = case
           when p.type = 'indie_game_developer' and public._intel_is_protected_indie_entity(p.name, coalesce(nullif(p.project_title, ''), p.facts #>> '{verified,project_title}', p.facts #>> '{verified,game}')) then 'rejected'
           when p.type = 'indie_game_developer' and (coalesce(p_estimates #>> '{suppression,suppressed}', 'false') = 'true'
             or coalesce(p_estimates #>> '{suppression,bounced}', 'false') = 'true'
             or coalesce(p_estimates #>> '{suppression,complained}', 'false') = 'true') then 'suppressed'
           when status in ('suppressed','lead_linked','contacted','replied','closed') then status
           else p_decision end,
         review_reason = left(coalesce(p_reason, ''), 300),
         facts = case when p.type = 'indie_game_developer' then
           jsonb_set(
             jsonb_set(facts, '{estimates}', coalesce(facts -> 'estimates', '{}'::jsonb) || coalesce(p_estimates, '{}'::jsonb)),
             '{verified}', coalesce(facts -> 'verified', '{}'::jsonb) || jsonb_build_object(
               'contactability', coalesce(p_estimates -> 'contactability', facts #> '{verified,contactability}', '{}'::jsonb)), true)
           else jsonb_set(facts, '{estimates}', coalesce(p_estimates, '{}'::jsonb)) end,
         updated_at = now()
   where id = p_prospect_id and organization_id = v_org;
end $$;

-- The admin review button is not a way around mandatory gates or the 75-point bar.
-- Linking an already-existing CRM lead remains available for duplicate resolution.
create or replace function public.intel_admin_review_prospect(p_prospect_id uuid, p_action text, p_lead_id uuid default null) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  p public.intel_prospects;
  s public.intel_settings;
  v_lead uuid;
  v_gates jsonb;
  v_gate_names text[] := array['project_identity','active_or_marketing_opportunity','anime_visual_identity','verified_usable_email','suppression_clear','protected_account_clear','sufficient_evidence'];
  v_gate text;
  v_email text;
begin
  select * into p from public.intel_prospects where id = p_prospect_id and organization_id = v_org for update;
  if not found then raise exception 'prospect not found'; end if;
  select * into s from public.intel_settings where organization_id = v_org;
  if p.type = 'indie_game_developer' and p_action <> 'reject' then
    if public._intel_is_protected_indie_entity(p.name, coalesce(nullif(p.project_title, ''), p.facts #>> '{verified,project_title}', p.facts #>> '{verified,game}')) then
      raise exception 'protected Studio Kairegi account cannot be approved or linked';
    end if;
    if p.status = 'suppressed'
       or coalesce(p.facts #>> '{verified,bounced}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,complained}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,complaint}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,unsubscribed}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,do_not_contact}', 'false') = 'true'
       or coalesce(p.facts #>> '{verified,suppressed}', 'false') = 'true'
       or exists (select 1 from public.intel_suppressions x where x.organization_id = v_org
         and ((x.kind = 'email' and x.value = public.intel_norm_email(p.contact_email) and p.contact_email <> '')
           or (x.kind = 'domain' and (x.value = public._intel_indie_domain(p.domain) or public.intel_norm_email(p.contact_email) like '%@' || x.value)))) then
      raise exception 'suppressed indie prospect cannot be approved or linked';
    end if;
  end if;
  if p_action in ('approve','create_new','use_lead') and p.type = 'indie_game_developer' then
    if coalesce(p.facts #>> '{estimates,qualification_v2,model}', '') <> 'indie-project-v2' then
      raise exception 'indie V2 qualification model version is required';
    end if;
    v_gates := p.facts #> '{estimates,qualification_v2,mandatory_gates}';
    if v_gates is null then raise exception 'indie V2 qualification has not been completed'; end if;
    foreach v_gate in array v_gate_names loop
      if coalesce(v_gates -> v_gate ->> 'passed', 'false') <> 'true' then
        raise exception 'indie V2 mandatory gate failed: %', v_gate;
      end if;
    end loop;
    if p.score is null or p.score < 75 then raise exception 'indie V2 prospect must score at least 75 before approval'; end if;
    if public._intel_is_protected_indie_entity(p.name, coalesce(nullif(p.project_title, ''), p.facts #>> '{verified,project_title}', p.facts #>> '{verified,game}')) then raise exception 'protected Studio Kairegi account cannot be approved'; end if;
    if coalesce(p.facts #>> '{estimates,qualification_v2,contactability,status}', p.facts #>> '{estimates,contactability,status}', '') <> 'pass' then
      raise exception 'indie V2 contactability must pass before approval';
    end if;
    v_email := public.intel_norm_email(p.contact_email);
    if exists (select 1 from public.intel_suppressions x where x.organization_id = v_org and
      ((x.kind = 'email' and x.value = v_email and v_email <> '') or (x.kind = 'domain' and (x.value = public._intel_indie_domain(p.domain) or v_email like '%@' || x.value)))) then
      raise exception 'indie V2 prospect is suppressed';
    end if;
  end if;
  if p_action = 'approve' then
    update public.intel_prospects set status = 'qualified', review_reason = '', updated_at = now() where id = p.id;
  elsif p_action = 'reject' then
    if p.type = 'indie_game_developer' and p.status = 'suppressed' then
      update public.intel_prospects set review_reason = 'Suppression preserved; prospect remains suppressed', updated_at = now() where id = p.id;
    else
      update public.intel_prospects set status = 'rejected', updated_at = now() where id = p.id;
    end if;
  elsif p_action = 'use_lead' then
    if not exists (select 1 from public.leads l where l.id = p_lead_id and l.user_id = s.owner_user_id) then raise exception 'lead not found'; end if;
    update public.intel_prospects set lead_id = p_lead_id, status = 'lead_linked', review_reason = '', updated_at = now() where id = p.id;
    v_lead := p_lead_id;
  elsif p_action = 'create_new' then
    -- Force the same database qualification trigger used by the normal worker
    -- path before any new CRM lead is created.
    if p.type = 'indie_game_developer' then
      update public.intel_prospects set status = 'qualified', updated_at = now() where id = p.id;
    end if;
    v_lead := public._intel_create_ai_lead(v_org, p.id, auth.uid());
  else
    raise exception 'unknown action';
  end if;
  perform public._intel_log(v_org, 'admin', 'review_prospect_' || p_action, 'prospect', p.id::text, jsonb_build_object('lead_id', v_lead));
  return jsonb_build_object('ok', true, 'lead_id', v_lead);
end $$;

create or replace function public.intel_agent_indie_suppression_facts(p_email text default '', p_domain text default '')
returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  v_email text := public.intel_norm_email(p_email);
  v_domain text := public._intel_indie_domain(p_domain);
  v_bounced boolean;
  v_complained boolean;
  v_suppressed boolean;
  v_reason text;
begin
  select
    coalesce(bool_or(lower(reason) like '%bounce%'), false),
    coalesce(bool_or(lower(reason) like '%complaint%'), false),
    coalesce(bool_or(true), false),
    string_agg(distinct nullif(reason, ''), '; ')
  into v_bounced, v_complained, v_suppressed, v_reason
  from public.intel_suppressions
  where organization_id = v_org
    and (
      (kind = 'email' and value = v_email and v_email <> '')
      or (kind = 'domain' and (value = v_domain or v_email like '%@' || value))
    );
  return jsonb_build_object('bounced', v_bounced, 'complained', v_complained,
    'suppressed', v_suppressed, 'suppression_reason', coalesce(v_reason, ''));
end $$;

-- Keep one prospect per game project rather than one per developer-domain.
-- A developer may have several projects; publisher identity is stored separately in facts.
create or replace function public.intel_agent_upsert_prospect(
  p_type text, p_name text, p_website text, p_contact_name text, p_contact_email text,
  p_contact_source_url text, p_country text, p_language text, p_verified jsonb default '{}'::jsonb)
returns jsonb language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  v_domain text := public.intel_domain_of(p_website);
  v_key text;
  v_id uuid;
  v_new boolean := false;
  v_camp uuid;
  v_email text := public.intel_norm_email(p_contact_email);
  v_supp boolean := false;
  v_title text := nullif(btrim(coalesce(nullif(p_verified ->> 'project_title', ''), p_verified ->> 'game', '')), '');
begin
  if p_type not in ('indie_game_developer','animation_studio_support') then raise exception 'bad type'; end if;
  if btrim(coalesce(p_name, '')) = '' then raise exception 'name required'; end if;

  if p_type = 'indie_game_developer' and v_title is not null then
    -- The same developer/title pair deduplicates Steam and itch discoveries. This key is
    -- project-scoped, not publisher-scoped or developer-domain-scoped.
    v_key := 'project:' || md5(lower(regexp_replace(btrim(p_name), '\s+', ' ', 'g')) || '|' || lower(regexp_replace(btrim(v_title), '\s+', ' ', 'g')));
  elsif p_type = 'indie_game_developer' and coalesce(p_verified ->> 'steam_appid', '') ~ '^[0-9]+$' then
    v_key := 'project:steam:' || (p_verified ->> 'steam_appid');
  elsif p_type = 'indie_game_developer' and coalesce(p_verified ->> 'itch_project_id', '') <> '' then
    v_key := 'project:itch:' || md5(lower(p_verified ->> 'itch_project_id'));
  else
    v_key := case when v_domain <> '' then 'd:' || v_domain else 'n:' || public.intel_norm_company(p_name) end;
  end if;
  if v_key in ('d:', 'n:', 'project:') then raise exception 'cannot derive a canonical key'; end if;

  select id into v_camp from public.intel_campaigns where organization_id = v_org and type = p_type;
  v_supp := exists (select 1 from public.intel_suppressions x where x.organization_id = v_org and
    ((x.kind = 'email' and x.value = v_email and v_email <> '')
      or (x.kind = 'domain' and x.value = v_domain and v_domain <> '')
      or (x.kind = 'domain' and v_email like '%@' || x.value)));

  insert into public.intel_prospects(organization_id, campaign_id, type, canonical_key, name, website, domain, contact_name,
    contact_email, contact_source_url, country, language, facts, status)
  values (v_org, v_camp, p_type, v_key, left(p_name, 200), left(coalesce(p_website, ''), 300), v_domain,
    left(coalesce(p_contact_name, ''), 200), v_email, left(coalesce(p_contact_source_url, ''), 500),
    left(coalesce(p_country, ''), 80), left(coalesce(nullif(p_language, ''), 'en'), 12),
    case when p_type = 'indie_game_developer' then
      jsonb_build_object('verified', coalesce(p_verified, '{}'::jsonb) || jsonb_build_object(
        'project_source_urls', case when coalesce(p_verified ->> 'project_url', '') <> '' then jsonb_build_array(p_verified ->> 'project_url') else '[]'::jsonb end,
        'last_discovered_at', coalesce(p_verified ->> 'discovered_at', now()::text)), 'estimates', '{}'::jsonb)
      else jsonb_build_object('verified', coalesce(p_verified, '{}'::jsonb), 'estimates', '{}'::jsonb) end,
    case when v_supp then 'suppressed' else 'discovered' end)
  on conflict (organization_id, canonical_key) do update
    set website = case when p_type = 'indie_game_developer' and public.intel_prospects.website = '' then excluded.website else public.intel_prospects.website end,
        domain = case when p_type = 'indie_game_developer' and public.intel_prospects.domain = '' then excluded.domain else public.intel_prospects.domain end,
        contact_email = case when public.intel_prospects.contact_email = '' then excluded.contact_email else public.intel_prospects.contact_email end,
        contact_name = case when public.intel_prospects.contact_name = '' then excluded.contact_name else public.intel_prospects.contact_name end,
        contact_source_url = case when public.intel_prospects.contact_source_url = '' then excluded.contact_source_url else public.intel_prospects.contact_source_url end,
        facts = case when p_type = 'indie_game_developer' then
          jsonb_set(public.intel_prospects.facts, '{verified}',
            coalesce(public.intel_prospects.facts -> 'verified', '{}'::jsonb) || coalesce(p_verified, '{}'::jsonb) ||
            jsonb_build_object('discovered_at', coalesce(public.intel_prospects.facts #>> '{verified,discovered_at}', p_verified ->> 'discovered_at'),
              'last_discovered_at', coalesce(p_verified ->> 'discovered_at', now()::text), 'project_source_urls',
              case when coalesce(p_verified ->> 'project_url', '') <> ''
                and not (coalesce(public.intel_prospects.facts #> '{verified,project_source_urls}', '[]'::jsonb) @> jsonb_build_array(p_verified ->> 'project_url'))
                then coalesce(public.intel_prospects.facts #> '{verified,project_source_urls}', '[]'::jsonb) || jsonb_build_array(p_verified ->> 'project_url')
                else coalesce(public.intel_prospects.facts #> '{verified,project_source_urls}', '[]'::jsonb) end), true)
          else jsonb_set(public.intel_prospects.facts, '{verified}',
            coalesce(public.intel_prospects.facts -> 'verified', '{}'::jsonb) || coalesce(p_verified, '{}'::jsonb), true) end,
        updated_at = now()
  returning id, (xmax = 0) into v_id, v_new;

  if p_type = 'indie_game_developer' and v_supp then
    update public.intel_prospects set status = 'suppressed', review_reason = 'Existing email/domain suppression conflict'
      where id = v_id and organization_id = v_org and status not in ('lead_linked','contacted','replied','closed');
  end if;
  if p_type = 'indie_game_developer' and public._intel_is_protected_indie_entity(p_name, v_title) then
    update public.intel_prospects set status = 'rejected', review_reason = 'Protected Studio Kairegi account'
      where id = v_id and organization_id = v_org and status not in ('lead_linked','contacted','replied','closed');
  end if;
  return jsonb_build_object('prospect_id', v_id, 'created', v_new, 'suppressed', v_supp);
end $$;

-- Full agent-only source status is separate from intel_agent_sources(), which intentionally
-- returns only sources with remaining budget. Paginated discovery needs to distinguish an
-- exhausted daily budget from a disabled/unconfirmed source so it can retain the cursor.
create or replace function public.intel_agent_source_status() returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_agent_org(); v_today date := public._intel_today(v_org);
begin
  update public.intel_sources set used_today = 0, used_on = v_today
    where organization_id = v_org and used_on is distinct from v_today;
  return coalesce((select jsonb_object_agg(key, jsonb_build_object(
    'enabled', enabled,
    'tos_status', tos_status,
    'remaining', greatest(daily_budget - used_today, 0),
    'circuit_open', circuit_open_until is not null and circuit_open_until > now(),
    'retry_after_seconds', case when circuit_open_until is not null and circuit_open_until > now()
      then greatest(60, ceil(extract(epoch from (circuit_open_until - now())))::integer) else 0 end
  )) from public.intel_sources where organization_id = v_org), '{}'::jsonb);
end $$;

-- A daily initial scan must not race a queued continuation cursor. This lets the existing
-- durable intel_jobs queue carry the next page across worker restarts and source-budget resets.
create or replace function public.intel_agent_discovery_has_pending(p_type text) returns boolean
language sql stable security definer set search_path = pg_catalog, public as $$
  select case when p_type <> 'indie_game_developer' then false else exists (
    select 1 from public.intel_jobs j
    where j.organization_id = public.intel_agent_org()
      and j.kind = 'discover'
      and j.state in ('queued', 'running')
      and j.payload ->> 'type' = p_type
      and j.payload ? 'cursor'
      and jsonb_typeof(j.payload -> 'cursor') = 'object'
      and j.payload -> 'cursor' <> '{}'::jsonb
  ) end
$$;

revoke all on function public.intel_agent_source_status() from public, anon;
grant execute on function public.intel_agent_source_status() to authenticated;
revoke all on function public.intel_agent_discovery_has_pending(text) from public, anon;
grant execute on function public.intel_agent_discovery_has_pending(text) to authenticated;

commit;

-- Admin submits the URL and metadata they personally observed. We persist the intake and enqueue the existing durable
-- manual_post job. No Kickstarter URL is fetched, parsed, crawled, or sent to an external search provider.
create or replace function public.intel_admin_add_kickstarter_project(
  p_url text, p_title text, p_creator text, p_developer text default '', p_website text default '', p_notes text default '',
  p_permission_reference text default '', p_permission_confirmed boolean default false
) returns uuid
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  v_id uuid;
  v_url text := btrim(coalesce(p_url, ''));
  v_title text := left(btrim(coalesce(p_title, '')), 200);
  v_creator text := left(btrim(coalesce(p_creator, '')), 160);
  v_developer text := left(btrim(coalesce(p_developer, '')), 160);
  v_website text := left(btrim(coalesce(p_website, '')), 300);
  v_permission_reference text := left(btrim(coalesce(p_permission_reference, '')), 200);
  v_meta jsonb;
begin
  if p_permission_confirmed is not true or v_permission_reference = '' then
    raise exception 'prior written permission for commercial use is required; provide an internal permission reference';
  end if;
  if not exists (select 1 from public.intel_campaigns c where c.organization_id = v_org and c.type = 'indie_game_developer') then
    raise exception 'Indie Game Developers campaign is not configured';
  end if;
  if exists (select 1 from public.intel_sources s where s.organization_id = v_org and s.key = 'kickstarter' and s.tos_status = 'excluded') then
    raise exception 'Kickstarter intake is excluded by administrator';
  end if;
  if v_url !~ '^https://(www\.)?kickstarter\.com/projects/[A-Za-z0-9_-]{1,100}/[A-Za-z0-9_-]{1,160}/?$' then
    raise exception 'use a canonical HTTPS Kickstarter project URL';
  end if;
  if v_title = '' or v_creator = '' then raise exception 'project title and creator name are required'; end if;
  if v_website <> '' and (v_website !~ '^https?://[^/[:space:]]+([/][^[:space:]]*)?$' or lower(v_website) ~ 'kickstarter\.com') then
    raise exception 'developer website must be a separate official HTTP(S) website';
  end if;
  v_url := regexp_replace(v_url, '^https://(www\.)?kickstarter\.com', 'https://www.kickstarter.com');
  v_url := regexp_replace(v_url, '/+$', '') || '/';
  v_meta := jsonb_build_object(
    'project_title', v_title,
    'creator_name', v_creator,
    'developer_name', v_developer,
    'developer_website', v_website,
    'admin_notes', left(btrim(coalesce(p_notes, '')), 600),
    'discovery_method', 'administrator_manual_url_import',
    'developer_identity_status', 'admin_attributed_unverified',
    'kickstarter_creator_role', 'unverified',
    'qualification_evidence_status', 'not_assessed',
    'permission_confirmed_by_admin', true,
    'permission_reference', v_permission_reference,
    'permission_recorded_at', now()
  );
  insert into public.intel_source_items(organization_id, source_key, item_type, url, url_hash, title, snippet, meta)
  values (v_org, 'kickstarter_manual', 'game_project', v_url, md5(v_url), v_title,
    left(btrim(coalesce(p_notes, '')), 600), v_meta)
  on conflict (organization_id, url_hash) do update set
    last_seen = now(), title = excluded.title, snippet = excluded.snippet,
    meta = coalesce(public.intel_source_items.meta, '{}'::jsonb) || excluded.meta
  returning id into v_id;
  insert into public.intel_jobs(organization_id, kind, payload, idempotency_key)
  values (v_org, 'manual_post', jsonb_build_object('item_id', v_id, 'type', 'indie_game_developer'),
    'kickstarter_manual:' || v_id::text || ':' || md5(v_meta::text))
  on conflict (organization_id, idempotency_key) do nothing;
  perform public._intel_log(v_org, 'admin', 'add_kickstarter_project', 'source_item', v_id::text,
    jsonb_build_object('discovery_method', 'manual_url_import'));
  return v_id;
end $$;

revoke all on function public.intel_admin_add_kickstarter_project(text,text,text,text,text,text,text,boolean) from public, anon;
grant execute on function public.intel_admin_add_kickstarter_project(text,text,text,text,text,text,text,boolean) to authenticated;

commit;

-- Indie Game Developers V2 / Phase 5: evidence-backed identity, visual review and contact verification.
-- No automated Kickstarter access. Admin attestations must cite exact, already-reviewed evidence rows.
begin;

create or replace function public.intel_admin_add_indie_evidence(
  p_prospect_id uuid, p_fact_type text, p_source_url text, p_snippet text,
  p_event_date date default null, p_media_url text default ''
) returns uuid
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  p public.intel_prospects;
  v_id uuid;
  v_type text := lower(btrim(coalesce(p_fact_type,'')));
  v_url text := btrim(coalesce(p_source_url,''));
  v_media text := btrim(coalesce(p_media_url,''));
  v_hash text;
  v_verified jsonb;
  v_milestones jsonb;
  v_allowed text[] := array[
    'developer_identity','project_identity','contact_email','anime_style_description','official_anime_style_description',
    'anime_visual_screenshot','anime_visual_artwork','anime_visual_trailer','anime_character_art',
    'devlog','development_log','development_update','developer_progress_update','project_update',
    'demo_update','playtest_update','build_update','patch_update','project_announcement',
    'marketing_opportunity','dlc_announcement','major_update_announcement','relaunch_announcement','console_release_announcement'
  ];
  v_date_types text[] := array['devlog','development_log','development_update','developer_progress_update','project_update',
    'demo_update','playtest_update','build_update','patch_update','project_announcement','marketing_opportunity',
    'dlc_announcement','major_update_announcement','relaunch_announcement','console_release_announcement'];
  v_domain text;
begin
  select * into p from public.intel_prospects where id=p_prospect_id and organization_id=v_org and type='indie_game_developer' for update;
  if not found then raise exception 'indie prospect not found'; end if;
  if v_type <> all(v_allowed) then raise exception 'unsupported Indie evidence type'; end if;
  if v_url !~ '^https?://' or v_url ~* 'kickstarter\.com' then raise exception 'evidence must come from a reviewed non-Kickstarter HTTP(S) source'; end if;
  if length(btrim(coalesce(p_snippet,''))) < 20 or length(p_snippet) > 1800 then raise exception 'evidence description must be between 20 and 1800 characters'; end if;
  v_domain := public._intel_indie_domain(v_url);
  if v_domain = '' or not (
    (p.website <> '' and v_domain = public._intel_indie_domain(p.website))
    or (p.project_source_url <> '' and p.project_source_url !~* 'kickstarter\.com' and v_domain = public._intel_indie_domain(p.project_source_url))
  ) then raise exception 'evidence URL must match the prospect official website or non-Kickstarter project source'; end if;
  if v_type = any(v_date_types) and p_event_date is null then raise exception 'dated evidence requires an explicit event date'; end if;
  if v_type in ('devlog','development_log','development_update','developer_progress_update','project_update','demo_update','playtest_update','build_update','patch_update','project_announcement') and p_event_date > current_date then
    raise exception 'activity and announcement dates cannot be in the future';
  end if;
  if v_type in ('anime_visual_screenshot','anime_visual_artwork','anime_visual_trailer','anime_character_art') then
    if v_media !~ '^https?://' or v_media ~* 'kickstarter\.com' then raise exception 'visual review requires an HTTP(S) media URL'; end if;
  end if;
  if v_type in ('anime_style_description','official_anime_style_description') and p_snippet !~* '(anime|manga).{0,80}(style|inspired|art|visual|aesthetic|character)|(style|art|visual|aesthetic|character).{0,80}(anime|manga)' then
    raise exception 'style evidence must explicitly describe anime/manga visual style';
  end if;
  if v_type='developer_identity' and p_snippet !~* '[[:alnum:]]' then raise exception 'developer identity evidence is required'; end if;
  v_hash := md5(v_type || '|' || v_url || '|' || coalesce(p_snippet,'') || '|' || coalesce(p_event_date::text,'') || '|' || v_media);
  insert into public.intel_evidence(organization_id,prospect_id,source,url,fact_type,snippet,content_hash,verified)
  values(v_org,p.id,'admin_review',v_url,v_type,
    left(btrim(p_snippet) || case when v_media<>'' then ' | reviewed media: '||v_media else '' end,2000),v_hash,true)
  on conflict (prospect_id,fact_type,content_hash) do update set fetched_at=now()
  returning id into v_id;

  v_verified := coalesce(p.facts->'verified','{}'::jsonb);
  if v_type in ('devlog','development_log','development_update','developer_progress_update','project_update','demo_update','playtest_update','build_update','patch_update') then
    if p.project_latest_activity_date is null or p_event_date > p.project_latest_activity_date then
      v_verified := jsonb_set(v_verified,'{project_latest_activity_date}',to_jsonb(p_event_date::text),true);
      update public.intel_prospects set project_latest_activity_date=p_event_date where id=p.id;
    end if;
  elsif v_type='project_announcement' then
    if p.project_announcement_date is null or p_event_date < p.project_announcement_date then
      v_verified := jsonb_set(v_verified,'{project_announcement_date}',to_jsonb(p_event_date::text),true);
      update public.intel_prospects set project_announcement_date=p_event_date where id=p.id;
    end if;
  elsif v_type in ('marketing_opportunity','dlc_announcement','major_update_announcement','relaunch_announcement','console_release_announcement') then
    v_milestones := coalesce(p.facts #> '{verified,project_marketing_milestones}',p.project_marketing_milestones,'[]'::jsonb);
    if not exists (select 1 from jsonb_array_elements(v_milestones) m where m->>'date'=p_event_date::text and m->>'source_url'=v_url and m->>'kind'=v_type) then
      v_milestones := v_milestones || jsonb_build_array(jsonb_build_object(
        'date',p_event_date::text,'source_url',v_url,'kind',v_type,'verified',true,'evidence_id',v_id::text));
    end if;
    v_verified := jsonb_set(v_verified,'{project_marketing_milestones}',v_milestones,true);
    update public.intel_prospects set project_marketing_milestones=v_milestones where id=p.id;
  end if;
  update public.intel_prospects set facts=jsonb_set(facts,'{verified}',v_verified,true),updated_at=now() where id=p.id;
  perform public._intel_log(v_org,'admin','add_indie_evidence','prospect',p.id::text,jsonb_build_object('fact_type',v_type,'evidence_id',v_id,'source_url',v_url));
  return v_id;
end $$;

create or replace function public.intel_admin_verify_indie_identity(

create or replace function public.intel_admin_verify_indie_contact(
  p_prospect_id uuid, p_email text, p_evidence_id uuid
) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  p public.intel_prospects;
  e public.intel_evidence;
  v_email text := public.intel_norm_email(p_email);
  v_domain text;
  v_verified jsonb;
begin
  select * into p from public.intel_prospects where id=p_prospect_id and organization_id=v_org and type='indie_game_developer' for update;
  if not found then raise exception 'indie prospect not found'; end if;
  if public._intel_is_protected_indie_entity(p.name,p.project_title) then raise exception 'protected Studio Kairegi account contact cannot be verified'; end if;
  if v_email='' or v_email !~ '^[^@[:space:]]+@[^@[:space:]]+\.[^@[:space:]]+$' then raise exception 'valid exact email address is required'; end if;

  if e.id is null or position(v_email in lower(e.snippet))=0 then raise exception 'reviewed contact evidence must contain the exact email address'; end if;
  v_domain := public._intel_indie_domain(p.website);
  if v_domain='' or public._intel_indie_domain(e.url)<>v_domain then raise exception 'contact email must be documented on the official developer website'; end if;
  if p.status='suppressed' or coalesce(p.facts #>> '{verified,suppressed}','false')='true' or coalesce(p.facts #>> '{verified,unsubscribed}','false')='true' or coalesce(p.facts #>> '{verified,do_not_contact}','false')='true' or coalesce(p.facts #>> '{verified,bounced}','false')='true' or coalesce(p.facts #>> '{verified,complained}','false')='true' or coalesce(p.facts #>> '{verified,complaint}','false')='true' then

-- Indie Game Developers V2 / Phase 4 corrective fixes.
-- 1) Excluding Kickstarter cancels queued/running manual intake jobs.
-- 2) Worker import is transactionally gated against source exclusion.
-- 3) Kickstarter's unverified attribution cannot downgrade verified identity facts.
begin;

-- Preserve the existing admin source controls while making exclusion an immediate
-- queue action. Running workers are also cancelled; the transactional commit RPC
-- below prevents a worker that has not committed yet from importing after exclusion.
create or replace function public.intel_admin_update_source(
  p_key text, p_enabled boolean default null, p_tos_status text default null, p_daily_budget int default null
) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  s public.intel_sources;
  v_excluded boolean := false;
  v_cancelled integer := 0;
begin
  select * into s from public.intel_sources
   where organization_id = v_org and key = p_key for update;
  if not found then raise exception 'unknown source'; end if;
  if p_key = 'kickstarter' and coalesce(p_enabled, s.enabled) is true then
    raise exception 'automated Kickstarter discovery is unavailable; use manual project intake until written permission and an authorized connector exist';
  end if;
  if p_tos_status is not null and p_tos_status not in ('confirmed','unverified','excluded') then
    raise exception 'bad terms status';
  end if;
  if coalesce(p_tos_status, s.tos_status) <> 'confirmed' and coalesce(p_enabled, s.enabled) then
    raise exception 'Confirm the source''s terms before enabling it';
  end if;

  update public.intel_sources set
    enabled = coalesce(p_enabled, enabled),
    tos_status = coalesce(p_tos_status, tos_status),
    daily_budget = coalesce(p_daily_budget, daily_budget),
    consecutive_failures = case when p_enabled is true then 0 else consecutive_failures end,
    circuit_open_until = case when p_enabled is true then null else circuit_open_until end,
    updated_at = now()
  where organization_id = v_org and key = p_key
  returning * into s;

  v_excluded := p_key = 'kickstarter' and s.tos_status = 'excluded';
  if v_excluded then
    update public.intel_jobs j
       set state = 'cancelled', finished_at = now(), updated_at = now(),
           lease_until = null, locked_by = null,
           last_error = 'Cancelled: Kickstarter intake excluded by administrator'
     where j.organization_id = v_org
       and j.kind = 'manual_post'
       and j.state in ('queued','running')
       and exists (
         select 1 from public.intel_source_items i
          where i.organization_id = v_org
            and i.id::text = j.payload ->> 'item_id'
            and i.source_key = 'kickstarter_manual'
       );
    get diagnostics v_cancelled = row_count;
    update public.intel_source_items i
       set meta = coalesce(i.meta, '{}'::jsonb) || jsonb_build_object(
         'intake_status', 'blocked_source_excluded',
         'intake_blocked_at', now()
       ), last_seen = now()
     where i.organization_id = v_org
       and i.source_key = 'kickstarter_manual'
       and exists (
         select 1 from public.intel_jobs j
          where j.organization_id = v_org
            and j.kind = 'manual_post'
            and j.state = 'cancelled'
            and j.payload ->> 'item_id' = i.id::text
            and j.last_error = 'Cancelled: Kickstarter intake excluded by administrator'
       );
  end if;

  perform public._intel_log(v_org, 'admin', 'update_source', 'source', p_key,
    jsonb_build_object('enabled', p_enabled, 'tos', p_tos_status, 'budget', p_daily_budget,
                       'kickstarter_jobs_cancelled', v_cancelled));
  return to_jsonb(s) || jsonb_build_object('cancelled_jobs', v_cancelled);
end $$;

-- Atomic import boundary. The shared lock serializes the complete import against
-- intel_admin_update_source(), which takes an update lock on the same source row.
-- Kickstarter stays disabled for automated discovery; manual imports are permitted
-- unless the administrator explicitly marks the source excluded.
create or replace function public.intel_agent_kickstarter_commit_import(
  p_item_id uuid,
  p_name text,
  p_website text,
  p_verified jsonb,
  p_title text,
  p_notes text default ''
) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  v_source public.intel_sources;
  v_item public.intel_source_items;
  v_existing_verified jsonb := '{}'::jsonb;
  v_preserve jsonb := '{}'::jsonb;
  v_key text;
  v_status text;
  v_upsert jsonb;
  v_pid uuid;
begin
  select * into v_source
    from public.intel_sources
   where organization_id = v_org and key = 'kickstarter'
   for share;
  if not found or v_source.tos_status = 'excluded' then
    return jsonb_build_object('blocked', true, 'reason', 'source_excluded_or_unconfigured');
  end if;

  select * into v_item
    from public.intel_source_items
   where id = p_item_id and organization_id = v_org
   for update;
  if not found or v_item.source_key <> 'kickstarter_manual' or v_item.item_type <> 'game_project' then
    raise exception 'valid Kickstarter manual project item not found';
  end if;

  -- The canonical project key matches intel_agent_upsert_prospect() for this
  -- project identity, allowing us to snapshot existing verification before merge.
  if nullif(btrim(coalesce(p_title, '')), '') is not null then
    v_key := 'project:' || md5(
      lower(regexp_replace(btrim(p_name), '\s+', ' ', 'g')) || '|' ||
      lower(regexp_replace(btrim(p_title), '\s+', ' ', 'g'))
    );
    select coalesce(facts -> 'verified', '{}'::jsonb)
      into v_existing_verified
      from public.intel_prospects
     where organization_id = v_org and canonical_key = v_key
     for update;
  end if;

  v_status := lower(coalesce(v_existing_verified ->> 'developer_identity_status', ''));
  if v_status in ('independently_verified','human_confirmed','verified') then
    v_preserve := v_preserve || jsonb_strip_nulls(jsonb_build_object(
      'developer_name', v_existing_verified -> 'developer_name',
      'developer_identity_status', v_existing_verified -> 'developer_identity_status',
      'developer_attribution_source', v_existing_verified -> 'developer_attribution_source'
    ));
  end if;
  if lower(coalesce(v_existing_verified ->> 'kickstarter_creator_role_verified', 'false')) = 'true'
     and nullif(btrim(coalesce(v_existing_verified ->> 'kickstarter_creator_role', '')), '') is not null then
    v_preserve := v_preserve || jsonb_strip_nulls(jsonb_build_object(
      'kickstarter_creator_role', v_existing_verified -> 'kickstarter_creator_role',
      'kickstarter_creator_role_verified', v_existing_verified -> 'kickstarter_creator_role_verified'
    ));
  end if;

  v_upsert := public.intel_agent_upsert_prospect(
    'indie_game_developer', p_name, coalesce(p_website, ''), '', '', '', '', 'other', coalesce(p_verified, '{}'::jsonb)
  );
  v_pid := (v_upsert ->> 'prospect_id')::uuid;

  -- Restore independently verified identity/role facts over the unverified
  -- Kickstarter submission. Other project-specific evidence is still merged.
  if v_preserve <> '{}'::jsonb then
    update public.intel_prospects
       set facts = jsonb_set(
         coalesce(facts, '{}'::jsonb), '{verified}',
         coalesce(facts -> 'verified', '{}'::jsonb) || v_preserve, true
       ), updated_at = now()
     where id = v_pid and organization_id = v_org;
  end if;

  perform public.intel_agent_record_item(
    'kickstarter_manual', 'game_project', v_item.url, p_title, coalesce(p_notes, ''), '', v_pid,
    coalesce(p_verified, '{}'::jsonb) || jsonb_build_object('intake_status', 'imported')
  );

  return coalesce(v_upsert, '{}'::jsonb) || jsonb_build_object('blocked', false);
end $$;

revoke all on function public.intel_agent_kickstarter_commit_import(uuid,text,text,jsonb,text,text) from public, anon, authenticated;
grant execute on function public.intel_agent_kickstarter_commit_import(uuid,text,text,jsonb,text,text) to service_role;

commit;

  v_id uuid;
begin
  select * into p from public.intel_prospects
   where id=p_prospect_id and organization_id=v_org and type='indie_game_developer' for update;
  if not found then raise exception 'indie prospect not found'; end if;
  if v_url !~ '^https?://' or v_url ~* 'kickstarter\.com' then

  end if;
  v_domain := public._intel_indie_domain(v_url);

    raise exception 'suppressed, bounced, complained or opted-out contact cannot be recorded as eligible';
  end if;
  v_hash := md5('contact_email|'||v_url||'|'||v_snippet);
  insert into public.intel_evidence(organization_id,prospect_id,source,url,fact_type,snippet,content_hash,verified)

20261112192000_intelligence_indie_outreach_v6.sql
==================================================
-- Indie Game Developers V2 / Session 6.
-- Extends the existing approval-first sequence to two timed follow-ups and two evidence-gated re-engagements.
-- Existing organization limits are preserved; admins must deliberately raise max_followups to 4 to use all stages.
begin;

alter table public.intel_settings alter column max_followups set default 4;
alter table public.intel_settings drop constraint if exists intel_settings_max_followups_check;
alter table public.intel_settings add constraint intel_settings_max_followups_check check (max_followups between 0 and 4);

create or replace function public.intel_agent_followup_candidates(p_limit int default 20) returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  s public.intel_settings;
  v_today date;
begin
  select * into s from public.intel_settings where organization_id = v_org;
  v_today := public._intel_today(v_org);
  return coalesce((
    select jsonb_agg(x) from (
      select p.id as prospect_id,
             p.name,
             case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end as sent_at,
             exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id) as takeover,
             case
               when p.type = 'indie_game_developer' and f.sent_followups >= 2 then 'reengagement'
               when p.type = 'indie_game_developer' then 'followup'
               when not exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id)
                and f.sent_followups >= greatest(0, s.max_followups - 1) then 'reengagement'
               else 'followup'
             end as draft_kind
        from public.intel_prospects p
        join public.leads l on l.id = p.lead_id
        join lateral (
          select min(m.sent_at) filter (where m.kind = 'initial') as initial_sent_at,
                 max(m.sent_at) as last_sent_at
            from public.intel_messages m
           where m.prospect_id = p.id
             and m.direction = 'outbound'
             and m.kind in ('initial','followup')
             and m.state = 'sent'
        ) i on i.last_sent_at is not null
        cross join lateral (
          select count(*)::int as sent_followups
            from public.intel_messages f
           where f.prospect_id = p.id and f.kind = 'followup'
             and f.state not in ('discarded','failed')
        ) f
       where p.organization_id = v_org
         and p.status = 'contacted'
         and position('@' in public.intel_norm_email(l.email)) > 1
         and public.intel_norm_email(l.email) = public.intel_norm_email(p.contact_email)
         and case
           when p.type = 'indie_game_developer' then
             i.initial_sent_at is not null
             and f.sent_followups < least(4, s.max_followups)
             and case f.sent_followups
               when 0 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 5
               when 1 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 10
               when 2 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '3 months')::date)
               when 3 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '6 months')::date)
               else false
             end
           else
             public.intel_business_days_between(
               (greatest(i.last_sent_at, nullif(p.takeover ->> 'last_contact', '')::date::timestamp at time zone s.timezone) at time zone s.timezone)::date,
               v_today) >= s.followup_business_days
             and f.sent_followups < least(3, s.max_followups)
           end
         and not exists (
           select 1 from public.intel_messages f2
            where f2.prospect_id = p.id and f2.kind = 'followup'
              and f2.state in ('draft','pending_approval','approved','reserved','sending','unknown')
         )
         and not exists (
           select 1 from public.intel_messages r
            where r.prospect_id = p.id and r.direction = 'inbound'
         )
       order by case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end
       limit greatest(1, least(p_limit, 100))
    ) x
  ), '[]'::jsonb);
end $$;

revoke all on function public.intel_agent_followup_candidates(int) from public, anon;
grant execute on function public.intel_agent_followup_candidates(int) to authenticated;

comment on function public.intel_agent_followup_candidates(int) is
  'Indie sequence: business day 5 and 10 follow-ups, then month 3 and month 6 re-engagement. Any inbound reply interrupts the sequence. The worker additionally requires new verified dated project evidence before either re-engagement. Other campaign types retain their configured business-day sequence.';

notify pgrst, 'reload schema';
commit;

-- Enforce the campaign-specific follow-up ceiling at the final provider-send transition.
-- Indie Game Developers may use four follow-ups; every other campaign remains capped at three.
-- The Phase 8 mark_sending RPC locks intel_settings FOR UPDATE before it transitions a
-- message to `sending`, serializing this count with concurrent sends for the organization.
begin;

create or replace function public._intel_enforce_campaign_followup_send_cap()
returns trigger
language plpgsql
security definer
set search_path = pg_catalog, public
as $$
declare
  v_campaign_type text;
  v_configured_cap integer;
  v_effective_cap integer;
  v_existing integer;
begin
  if new.state <> 'sending' or new.direction <> 'outbound' or new.kind <> 'followup' then
    return new;
  end if;

  select p.type into v_campaign_type
    from public.intel_prospects p
   where p.id = new.prospect_id
     and p.organization_id = new.organization_id;
  if v_campaign_type is null then
    raise exception 'follow-up campaign type could not be verified'
      using errcode = '23514';
  end if;

  select s.max_followups into v_configured_cap
    from public.intel_settings s
   where s.organization_id = new.organization_id;
  if v_configured_cap is null then
    raise exception 'follow-up settings could not be verified'
      using errcode = '23514';
  end if;

  v_effective_cap := least(v_configured_cap,
    case when v_campaign_type = 'indie_game_developer' then 4 else 3 end);

  select count(*) into v_existing
    from public.intel_messages m
   where m.organization_id = new.organization_id
     and m.prospect_id = new.prospect_id
     and m.kind = 'followup'
     and m.id <> new.id
     and m.state in ('reserved', 'sending', 'sent', 'unknown');

  if v_existing >= v_effective_cap then
    raise exception 'campaign follow-up limit reached (maximum: %)', v_effective_cap
      using errcode = '23514';
  end if;

  return new;
end;
$$;

revoke all on function public._intel_enforce_campaign_followup_send_cap() from public, anon, authenticated;
drop trigger if exists intel_campaign_followup_send_cap on public.intel_messages;
create trigger intel_campaign_followup_send_cap
before update of state on public.intel_messages
for each row
when (new.state = 'sending' and old.state is distinct from new.state)
execute function public._intel_enforce_campaign_followup_send_cap();

create or replace function public._intel_chat_prepare(p_org uuid, p_tool text, p_args jsonb) returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  a jsonb := case when jsonb_typeof(coalesce(p_args, '{}'::jsonb)) = 'object' then p_args else '{}'::jsonb end;
  v_ids uuid[]; v_n int; v_have int; v_type text := nullif(a ->> 'type', ''); v_list text; v_val text; s public.intel_settings; v_clean jsonb;
  v_cap int; v_x text; pr public.intel_prospects; v_state text[];
begin
  select * into s from public.intel_settings where organization_id = p_org;
  if not found then return jsonb_build_object('ok', false, 'reason', 'not_set_up'); end if;

  if p_tool in ('approve_messages', 'discard_messages') then
    v_ids := public._intel_juuids(a, 'message_ids', 10);
    v_n := coalesce(array_length(v_ids, 1), 0);
    if v_n = 0 then return jsonb_build_object('ok', false, 'reason', 'no_valid_message_ids'); end if;
    v_state := case when p_tool = 'approve_messages' then array['pending_approval'] else array['draft', 'pending_approval', 'approved', 'failed'] end;
    select count(*), string_agg(format('- %s to %s, subject "%s"', m.kind, m.to_email, left(m.subject, 90)), E'\n' order by m.created_at)
      into v_have, v_list
      from public.intel_messages m
     where m.id = any (v_ids) and m.organization_id = p_org and m.direction = 'outbound' and m.state = any (v_state);
    if v_have <> v_n then return jsonb_build_object('ok', false, 'reason', 'some_messages_not_in_a_valid_state'); end if;
    if v_n = 1 and p_tool = 'approve_messages' then
      select E'\nStarts: ' || replace(left(m.body_text, 350), E'\n', ' ') into v_x from public.intel_messages m where m.id = v_ids[1];
    end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('message_ids', to_jsonb(v_ids)),
      'summary', case when p_tool = 'approve_messages' then format('Approve %s message(s) so the worker can send them (daily limits, warm-up and suppression still apply):', v_n)
                      else format('Discard %s message(s) (they will not be sent):', v_n) end || E'\n' || v_list || coalesce(v_x, ''));

  elsif p_tool = 'resume_campaign' then
    if v_type is null or v_type not in ('indie_game_developer', 'animation_studio_support') then return jsonb_build_object('ok', false, 'reason', 'bad_type'); end if;
    if not exists (select 1 from public.intel_campaigns where organization_id = p_org and type = v_type and status <> 'active') then return jsonb_build_object('ok', false, 'reason', 'campaign_already_active'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('type', v_type), 'summary', format('Start the "%s" campaign. Discovery runs again and approved messages for this campaign become sendable.', v_type));

  elsif p_tool = 'set_campaign_cap' then
    if v_type is null or v_type not in ('indie_game_developer', 'animation_studio_support') then return jsonb_build_object('ok', false, 'reason', 'bad_type'); end if;
    if (a ->> 'daily_cap') !~ '^\d{1,2}$' or (a ->> 'daily_cap')::int > 30 then return jsonb_build_object('ok', false, 'reason', 'daily_cap_must_be_0_to_30'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('type', v_type, 'daily_cap', (a ->> 'daily_cap')::int),
      'summary', format('Set the daily introduction limit for "%s" to %s.', v_type, a ->> 'daily_cap'));

  elsif p_tool = 'enable_sending' then
    if not s.kill_switch then return jsonb_build_object('ok', false, 'reason', 'sending_is_already_on'); end if;
    select count(*) into v_n from public.intel_messages where organization_id = p_org and state = 'approved';
    return jsonb_build_object('ok', true, 'args', '{}'::jsonb,
      'summary', format('Turn sending ON. %s message(s) are approved right now and will be emailed within the daily limits (%s introductions, %s follow-ups per day, warm-up %s). Needs a connected mailbox, a registered agent and your sender details.',
                        v_n, s.daily_initial_cap, s.daily_followup_cap, case when s.warmup_enabled then 'on' else 'off' end));

  elsif p_tool = 'set_autonomy' then
    v_x := a ->> 'autonomy';
    if v_x is null or v_x not in ('shadow', 'assisted', 'autonomous') then return jsonb_build_object('ok', false, 'reason', 'autonomy_must_be_shadow_assisted_or_autonomous'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('autonomy', v_x),
      'summary', 'Set autonomy to ' || v_x || ': ' || case v_x
        when 'shadow' then 'the agent drafts everything and every message waits for your approval.'
        when 'assisted' then 'follow-ups and safe auto-replies may go without approval; first introductions still need approval.'
        else 'introductions may be sent WITHOUT your approval (still within limits, suppression and the emergency stop).' end);

  elsif p_tool = 'update_limits' then
    v_clean := '{}'::jsonb;
    if a ? 'daily_initial_cap' then
      if (a ->> 'daily_initial_cap') !~ '^\d{1,2}$' or (a ->> 'daily_initial_cap')::int > 30 then return jsonb_build_object('ok', false, 'reason', 'daily_initial_cap_must_be_0_to_30'); end if;
      v_clean := v_clean || jsonb_build_object('daily_initial_cap', (a ->> 'daily_initial_cap')::int);
    end if;
    if a ? 'followup_business_days' then
      if (a ->> 'followup_business_days') !~ '^\d{1,2}$' or (a ->> 'followup_business_days')::int not between 1 and 30 then return jsonb_build_object('ok', false, 'reason', 'followup_business_days_must_be_1_to_30'); end if;
      v_clean := v_clean || jsonb_build_object('followup_business_days', (a ->> 'followup_business_days')::int);
    end if;
    if a ? 'max_followups' then
      if (a ->> 'max_followups') !~ '^\d$' or (a ->> 'max_followups')::int > 4 then return jsonb_build_object('ok', false, 'reason', 'max_followups_must_be_0_to_4'); end if;
      v_clean := v_clean || jsonb_build_object('max_followups', (a ->> 'max_followups')::int);
    end if;
    if v_clean = '{}'::jsonb then return jsonb_build_object('ok', false, 'reason', 'nothing_to_change'); end if;
    return jsonb_build_object('ok', true, 'args', v_clean, 'summary', 'Change sending limits: ' || (select string_agg(k || ' = ' || val, ', ') from jsonb_each_text(v_clean) as t(k, val)));

  elsif p_tool = 'add_suppression' then
    v_val := lower(btrim(coalesce(a ->> 'value', '')));
    if v_val !~ '^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$' and v_val !~ '^[a-z0-9-]+(\.[a-z0-9-]+)+$' then return jsonb_build_object('ok', false, 'reason', 'value_must_be_an_email_or_a_domain'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('value', v_val, 'reason', left(coalesce(a ->> 'reason', ''), 200)),
      'summary', format('Never contact %s again (%s).', v_val, case when position('@' in v_val) > 0 then 'this address' else 'everyone at this domain' end));

  elsif p_tool = 'review_prospect' then
    v_x := a ->> 'action';
    if v_x is null or v_x not in ('approve', 'reject') then return jsonb_build_object('ok', false, 'reason', 'action_must_be_approve_or_reject'); end if;
    select * into pr from public.intel_prospects where id = public._intel_juuid(a, 'prospect_id') and organization_id = p_org;
    if not found then return jsonb_build_object('ok', false, 'reason', 'prospect_not_found'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('prospect_id', pr.id, 'action', v_x),
      'summary', format('%s prospect "%s" (%s, currently %s).', case v_x when 'approve' then 'Approve' else 'Reject' end, pr.name, pr.type, pr.status));
  end if;
  return jsonb_build_object('ok', false, 'reason', 'unknown_tool');
end $$;

create or replace function public.intel_agent_classify_inbound(p_message_id uuid, p_class text, p_confidence numeric, p_escalate boolean, p_summary text, p_related_email text default null) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_agent_org(); m public.intel_messages; p public.intel_prospects; v_event text; v_title text; v_name text; v_who text;
begin
  if p_class not in ('positive','meeting_request','pricing','negotiation','portfolio_request','project_opportunity','rejection','unsubscribe','ambiguous','complaint','bounce','out_of_office','auto_reply','other','not_now','not_interested','already_has_trailer','wrong_person','referral','tell_me_more') then
    raise exception 'unknown class';
  end if;
  select * into m from public.intel_messages where id = p_message_id and organization_id = v_org and direction = 'inbound' for update;
  if not found then raise exception 'inbound message not found'; end if;
  update public.intel_messages set classification = p_class, classification_confidence = p_confidence, escalated = coalesce(p_escalate, false), summary = left(coalesce(p_summary, ''), 500) where id = m.id;
  if m.prospect_id is not null then select * into p from public.intel_prospects where id = m.prospect_id; end if;
  v_name := coalesce(p.name, m.from_email, 'a prospect');

  if p_class in ('unsubscribe','complaint') and m.from_email <> '' then
    insert into public.intel_suppressions(organization_id, kind, value, reason, source) values (v_org, 'email', m.from_email, p_class, 'agent') on conflict do nothing;
  end if;
  if p_class = 'bounce' and coalesce(p_related_email, '') <> '' then
    insert into public.intel_suppressions(organization_id, kind, value, reason, source) values (v_org, 'email', public.intel_norm_email(p_related_email), 'bounce', 'agent') on conflict do nothing;
    update public.intel_prospects set status = 'closed', updated_at = now() where organization_id = v_org and contact_email = public.intel_norm_email(p_related_email) and status in ('contacted','lead_linked');
  end if;
  if m.prospect_id is not null and p_class not in ('out_of_office','auto_reply','bounce') then
    update public.intel_prospects set status = case when p_class in ('rejection','unsubscribe','complaint','not_interested','already_has_trailer') then 'closed' else 'replied' end, updated_at = now() where id = m.prospect_id and status not in ('closed');
    -- stop anything still queued for this prospect
    update public.intel_messages set state = 'discarded', error = 'prospect replied' where prospect_id = m.prospect_id and direction = 'outbound' and kind in ('initial','followup') and state in ('draft','pending_approval','approved','reserved');
    if p_class in ('positive','meeting_request','pricing','negotiation','portfolio_request','project_opportunity') and m.lead_id is not null then
      update public.leads set stage = 'responded', stage_changed_at = now(), last_contacted_at = now() where id = m.lead_id and stage in ('pool','cold_email');
    end if;
  end if;

  v_event := case p_class when 'positive' then 'intel_positive_reply' when 'meeting_request' then 'intel_meeting_request' when 'pricing' then 'intel_pricing_inquiry'
                          when 'negotiation' then 'intel_negotiation' when 'portfolio_request' then 'intel_portfolio_request' when 'project_opportunity' then 'intel_project_opportunity'
                          when 'rejection' then 'intel_needs_review' when 'unsubscribe' then 'intel_unsubscribe' when 'bounce' then 'intel_bounce'
                          when 'ambiguous' then 'intel_needs_review' when 'complaint' then 'intel_needs_review'
                          when 'not_now' then 'intel_needs_review' when 'not_interested' then 'intel_needs_review'
                          when 'already_has_trailer' then 'intel_needs_review' when 'wrong_person' then 'intel_needs_review'
                          when 'referral' then 'intel_needs_review' when 'tell_me_more' then 'intel_needs_review' else null end;
  if v_event is not null then
    v_title := case p_class when 'positive' then 'Positive reply from ' when 'meeting_request' then 'Meeting request from ' when 'pricing' then 'Pricing enquiry from '
                            when 'negotiation' then 'Negotiation with ' when 'portfolio_request' then 'Portfolio request from ' when 'project_opportunity' then 'Project opportunity: '
                            when 'rejection' then 'Reply needs review from ' when 'unsubscribe' then 'Unsubscribe from ' when 'bounce' then 'Email bounced: '
                            when 'not_now' then 'Not now: ' when 'not_interested' then 'Not interested: '
                            when 'already_has_trailer' then 'Existing trailer: ' when 'wrong_person' then 'Wrong contact: '
                            when 'referral' then 'Referral: ' when 'tell_me_more' then 'More information requested by '
                            else 'Needs your review: ' end || v_name;
    perform public._intel_notify(v_org, v_event, v_title, coalesce(nullif(p_summary, ''), 'Open the conversation to read the message.'), m.lead_id, m.prospect_id, m.id, 'in:' || coalesce(m.zoho_message_id, m.id::text));
  end if;
  perform public._intel_log(v_org, 'agent', 'classify_inbound', 'message', m.id::text, jsonb_build_object('class', p_class, 'escalate', p_escalate));
  return jsonb_build_object('ok', true, 'notified', v_event is not null);
end $$;


create or replace function public.intel_agent_followup_candidates(p_limit int default 20) returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  s public.intel_settings;
  v_today date;
begin
  select * into s from public.intel_settings where organization_id = v_org;
  if not found then return '[]'::jsonb; end if;
  v_today := public._intel_today(v_org);
  return coalesce((
    select jsonb_agg(x) from (
      select p.id as prospect_id,
             p.name,
             case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end as sent_at,
             exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id) as takeover,
             case
               when p.type = 'indie_game_developer' and f.sent_followups >= 2 then 'reengagement'
               when p.type = 'indie_game_developer' then 'followup'
               when s.max_followups >= 2
                and not exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id)
                and f.sent_followups >= least(3, s.max_followups) - 1 then 'reengagement'
               else 'followup'
             end as draft_kind
        from public.intel_prospects p
        join public.leads l on l.id = p.lead_id
        join lateral (
          select min(m.sent_at) filter (where m.kind = 'initial') as initial_sent_at,
                 max(m.sent_at) as last_sent_at
            from public.intel_messages m
           where m.prospect_id = p.id and m.direction = 'outbound'
             and m.kind in ('initial','followup') and m.state = 'sent'
        ) i on i.last_sent_at is not null
        cross join lateral (
          select count(*)::int as sent_followups
            from public.intel_messages f
           where f.prospect_id = p.id and f.kind = 'followup'
             and f.state not in ('discarded','failed')
        ) f
       where p.organization_id = v_org
         and p.status = 'contacted'
         and coalesce(p.review_reason, '') !~ '^(Individualized English draft|Draft is too similar|Draft blocked by outbound policy|Re-engagement held:)'
         and position('@' in public.intel_norm_email(l.email)) > 1
         and public.intel_norm_email(l.email) = public.intel_norm_email(p.contact_email)
         and case
           when p.type = 'indie_game_developer' then
             i.initial_sent_at is not null
             and f.sent_followups < least(4, s.max_followups)
             and case f.sent_followups
               when 0 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 5
               when 1 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 10
               when 2 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '3 months')::date)
               when 3 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '6 months')::date)
               else false
             end
           else
             public.intel_business_days_between(
               (greatest(i.last_sent_at, nullif(p.takeover ->> 'last_contact', '')::date::timestamp at time zone s.timezone) at time zone s.timezone)::date,
               v_today) >= s.followup_business_days
             and f.sent_followups < least(3, s.max_followups)
           end
         and not exists (
           select 1 from public.intel_messages f2
            where f2.prospect_id = p.id and f2.kind = 'followup'
              and f2.state in ('draft','pending_approval','approved','reserved','sending','unknown')
         )
         and not exists (
           select 1 from public.intel_messages r
            where r.prospect_id = p.id and r.direction = 'inbound'
         )
       order by case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end
       limit greatest(1, least(coalesce(p_limit, 20), 100))
    ) x
  ), '[]'::jsonb);
end $$;

-- Phase 7 corrective safety patch: manual replies require explicit approval and
-- cannot be composed for closed/suppressed prospects or suppressed recipients.
-- Forward-only: do not edit historical migrations.
begin;

create or replace function public.intel_admin_compose_reply(p_prospect_id uuid, p_subject text, p_body text) returns uuid
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_admin_org();
  p public.intel_prospects;
  s public.intel_settings;
  v_in public.intel_messages;
  v_id uuid;
  v_to text;
begin
  select * into p
    from public.intel_prospects
   where id = p_prospect_id and organization_id = v_org
   for update;
  if not found then raise exception 'prospect not found'; end if;

  if p.status in ('closed', 'suppressed') then
    raise exception 'cannot compose a reply for a closed or suppressed prospect' using errcode = 'P0001';
  end if;

  select * into s from public.intel_settings where organization_id = v_org;
  select * into v_in
    from public.intel_messages
   where prospect_id = p.id and organization_id = v_org and direction = 'inbound'
   order by received_at desc nulls last, created_at desc
   limit 1;
  if not found then raise exception 'no inbound message exists for this prospect' using errcode = 'P0001'; end if;

  v_to := public.intel_norm_email(coalesce(nullif(v_in.from_email, ''), p.contact_email));
  if v_to = '' or position('@' in v_to) < 2 then
    raise exception 'recipient email is missing or invalid' using errcode = 'P0001';
  end if;

  if exists (
    select 1
      from public.intel_suppressions x
     where x.organization_id = v_org
       and ((x.kind = 'email' and x.value = v_to)
         or (x.kind = 'domain' and split_part(v_to, '@', 2) = x.value))
  ) then
    raise exception 'recipient is suppressed; reply was not created' using errcode = 'P0001';
  end if;

  insert into public.intel_messages(
    organization_id, prospect_id, lead_id, campaign_type, direction, kind,
    state, auto_sent, to_email, from_email, subject, body_text,
    idempotency_key, in_reply_to, references_ids
  ) values (
    v_org, p.id, p.lead_id, p.type, 'outbound', 'reply',
    'pending_approval', false, v_to, s.sender_email, left(coalesce(p_subject, ''), 300),
    left(coalesce(p_body, ''), 20000), 'admin-reply:' || gen_random_uuid()::text,
    v_in.rfc_message_id, v_in.references_ids
  ) returning id into v_id;

  perform public._intel_log(v_org, 'admin', 'compose_reply_pending_approval', 'message', v_id::text,
    jsonb_build_object('prospect_id', p.id, 'approval_required', true));
  return v_id;
end $$;

revoke all on function public.intel_admin_compose_reply(uuid, text, text) from public, anon;
grant execute on function public.intel_admin_compose_reply(uuid, text, text) to authenticated;

commit;

create or replace function public._intel_indie_domain_matches(p_host text, p_blocked_domain text)
returns boolean
language sql immutable
set search_path = pg_catalog, public
as $$
  select case
    when public._intel_indie_domain(p_host) = '' or public._intel_indie_domain(p_blocked_domain) = '' then false
    else public._intel_indie_domain(p_host) = public._intel_indie_domain(p_blocked_domain)
      or public._intel_indie_domain(p_host) like '%.' || public._intel_indie_domain(p_blocked_domain)
  end
$$;

-- Keep the worker's suppression lookup consistent with the contact-verification
-- RPC and include both recipient and prospect domains.
create or replace function public.intel_agent_indie_suppression_facts(p_email text default '', p_domain text default '')
returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  v_email text := public.intel_norm_email(p_email);
  v_email_domain text := split_part(v_email, '@', 2);
  v_domain text := public._intel_indie_domain(p_domain);
  v_bounced boolean;
  v_complained boolean;
  v_suppressed boolean;
  v_reason text;
begin
  select coalesce(bool_or(lower(reason) like '%bounce%'), false),
         coalesce(bool_or(lower(reason) like '%complaint%'), false),
         coalesce(bool_or(true), false),
         string_agg(distinct nullif(reason, ''), '; ')
    into v_bounced, v_complained, v_suppressed, v_reason
    from public.intel_suppressions s
   where s.organization_id = v_org
     and ((s.kind = 'email' and s.value = v_email and v_email <> '')
       or (s.kind = 'domain' and (public._intel_indie_domain_matches(v_domain, s.value)
         or public._intel_indie_domain_matches(v_email_domain, s.value))));
  return jsonb_build_object('bounced', v_bounced, 'complained', v_complained,
    'suppressed', v_suppressed, 'suppression_reason', coalesce(v_reason, ''));
end $$;

-- Last-resort database guard. The Phase 8 mark_sending RPC remains the primary
-- gate; this trigger prevents an Indie message from entering reserved/sending if
-- a parent-domain suppression (e.g. example.com for mail.example.com) exists.
create or replace function public._intel_indie_subdomain_send_guard()
returns trigger
language plpgsql security definer
set search_path = pg_catalog, public
as $$
declare
  v_org uuid := new.organization_id;
  v_email text := public.intel_norm_email(new.to_email);
  v_email_domain text := split_part(v_email, '@', 2);
  v_prospect_domain text := '';
  v_labels text[];
  v_suffix text;
  i integer;
begin
  if new.state not in ('reserved', 'sending') or new.prospect_id is null then return new; end if;
  if not exists (select 1 from public.intel_prospects p
    where p.id = new.prospect_id and p.organization_id = v_org and p.type = 'indie_game_developer') then
    return new;
  end if;
  select public._intel_indie_domain(p.domain) into v_prospect_domain
    from public.intel_prospects p where p.id = new.prospect_id and p.organization_id = v_org;

  -- Acquire the same policy-key locks used by suppression mutations for every
  -- parent suffix. This closes the race for a suppression inserted mid-send.
  v_labels := string_to_array(v_email_domain, '.');
  if coalesce(array_length(v_labels, 1), 0) >= 2 then
    for i in 1..array_length(v_labels, 1) - 1 loop
      v_suffix := array_to_string(v_labels[i:array_length(v_labels, 1)], '.');
      perform pg_advisory_xact_lock(hashtextextended(v_org::text || ':suppression:domain:' || v_suffix, 0));
    end loop;
  end if;
  if exists (select 1 from public.intel_suppressions s
      where s.organization_id = v_org
        and ((s.kind = 'email' and s.value = v_email)
          or (s.kind = 'domain' and (public._intel_indie_domain_matches(v_email_domain, s.value)
            or public._intel_indie_domain_matches(v_prospect_domain, s.value))))) then
    raise exception 'Indie campaign send blocked by email or parent-domain suppression';
  end if;
  return new;
end $$;
revoke all on function public._intel_indie_subdomain_send_guard() from public, anon, authenticated;
drop trigger if exists intel_indie_subdomain_send_guard on public.intel_messages;
create trigger intel_indie_subdomain_send_guard
before update of state on public.intel_messages
for each row when (new.state in ('reserved', 'sending'))
execute function public._intel_indie_subdomain_send_guard();

revoke all on function public._intel_indie_domain_matches(text,text) from public, anon, authenticated;
revoke all on function public.intel_agent_indie_suppression_facts(text,text) from public, anon;
grant execute on function public.intel_agent_indie_suppression_facts(text,text) to authenticated;

notify pgrst, 'reload schema';
commit;

create or replace function public.intel_agent_followup_candidates(p_limit int default 20) returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org();
  s public.intel_settings;
  v_today date;
begin
  select * into s from public.intel_settings where organization_id = v_org;
  v_today := public._intel_today(v_org);
  return coalesce((
    select jsonb_agg(x) from (
      select p.id as prospect_id,
             p.name,
             case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end as sent_at,
             exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id) as takeover,
             case
               when p.type = 'indie_game_developer' and f.sent_followups >= 2 then 'reengagement'
               when p.type = 'indie_game_developer' then 'followup'
               when not exists (select 1 from public.intel_lead_history h where h.prospect_id = p.id)
                and f.sent_followups >= greatest(0, s.max_followups - 1) then 'reengagement'
               else 'followup'
             end as draft_kind
        from public.intel_prospects p
        left join public.intel_campaigns campaign on campaign.id = p.campaign_id
          and campaign.organization_id = p.organization_id
        join public.leads l on l.id = p.lead_id
        join lateral (
          select min(m.sent_at) filter (where m.kind = 'initial') as initial_sent_at,
                 max(m.sent_at) as last_sent_at
            from public.intel_messages m
           where m.prospect_id = p.id
             and m.direction = 'outbound'
             and m.kind in ('initial','followup')
             and m.state = 'sent'
        ) i on i.last_sent_at is not null
        cross join lateral (
          select count(*)::int as sent_followups
            from public.intel_messages f
           where f.prospect_id = p.id and f.kind = 'followup'
             and f.state not in ('discarded','failed')
        ) f
       where p.organization_id = v_org
         and (p.type <> 'indie_game_developer' or campaign.status = 'active')
         and p.status = 'contacted'
         and position('@' in public.intel_norm_email(l.email)) > 1
         and public.intel_norm_email(l.email) = public.intel_norm_email(p.contact_email)
         and case
           when p.type = 'indie_game_developer' then
             i.initial_sent_at is not null
             and f.sent_followups < least(4, s.max_followups)
             and case f.sent_followups
               when 0 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 5
               when 1 then public.intel_business_days_between((i.initial_sent_at at time zone s.timezone)::date, v_today) >= 10
               when 2 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '3 months')::date)
               when 3 then v_today >= (((i.initial_sent_at at time zone s.timezone)::date + interval '6 months')::date)
               else false
             end
           else
             public.intel_business_days_between(
               (greatest(i.last_sent_at, nullif(p.takeover ->> 'last_contact', '')::date::timestamp at time zone s.timezone) at time zone s.timezone)::date,
               v_today) >= s.followup_business_days
             and f.sent_followups < least(3, s.max_followups)
           end
         and not exists (
           select 1 from public.intel_messages f2
            where f2.prospect_id = p.id and f2.kind = 'followup'
              and f2.state in ('draft','pending_approval','approved','reserved','sending','unknown')
         )
         and not exists (
           select 1 from public.intel_messages r
            where r.prospect_id = p.id and r.direction = 'inbound'
         )
       order by case when p.type = 'indie_game_developer' then i.initial_sent_at else i.last_sent_at end
       limit greatest(1, least(p_limit, 100))
    ) x
  ), '[]'::jsonb);
end $$;

-- Phase 4: job settlement hardening. Two small, additive function replacements. No table, column, constraint or data changes.
--
-- 1. intel_agent_fail_job now requires the job to be 'running'. Before this, it matched on (id, org, locked_by) only. complete_job leaves
--    locked_by set on a succeeded row, so a late or repeated failure report from the same worker (for example: the completion call committed,
--    its response was lost on the network, and the worker then reported a "transient" failure) turned a SUCCEEDED job back into 'queued' and
--    re-ran it. Now such a call returns 'not_owner' and changes nothing. fail_job_permanent already had this check.
-- 2. intel_agent_complete_job is idempotent for the same worker: repeating the completion of a job that this worker already completed returns
--    true instead of false. The worker retries completion on network errors; the first attempt may have committed.
--
-- The worker works with either version of both functions (it treats 'not_owner' and false as "lease lost" and never re-runs a handler for them).
-- Safe to apply twice (create or replace). Argument lists, return types and defaults are unchanged, so PostgREST resolution and grants are unaffected.
-- ROLLBACK (manual, only if needed): re-apply the two definitions from 20261109120000_intelligence_rpcs.sql.

create or replace function public.intel_agent_complete_job(p_job_id uuid, p_worker_id text, p_result jsonb default '{}'::jsonb) returns boolean
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_agent_org();
begin
  update public.intel_jobs set state = 'succeeded', result = p_result, lease_until = null, finished_at = now(), updated_at = now()
   where id = p_job_id and organization_id = v_org and state = 'running' and locked_by = p_worker_id;
  if found then return true; end if;
  -- Idempotent repeat by the same worker (the first response may have been lost): report success, change nothing.
  return exists (select 1 from public.intel_jobs
                  where id = p_job_id and organization_id = v_org and state = 'succeeded' and locked_by = p_worker_id);
end $$;

create or replace function public.intel_agent_fail_job(p_job_id uuid, p_worker_id text, p_error text, p_retry_seconds int default 60) returns text
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_agent_org(); j public.intel_jobs;
begin
  select * into j from public.intel_jobs
   where id = p_job_id and organization_id = v_org and state = 'running' and locked_by = p_worker_id for update;
  if not found then return 'not_owner'; end if;
  update public.intel_jobs
     set state = case when attempts >= max_attempts then 'dead' else 'queued' end,
         run_after = now() + make_interval(secs => greatest(1, least(p_retry_seconds, 86400))),
         last_error = left(p_error, 1000), locked_by = null, lease_until = null, updated_at = now(),
         finished_at = case when attempts >= max_attempts then now() end
   where id = j.id;
  return case when j.attempts >= j.max_attempts then 'dead' else 'queued' end;
end $$;

revoke all on function public.intel_agent_complete_job(uuid, text, jsonb) from public, anon;
grant execute on function public.intel_agent_complete_job(uuid, text, jsonb) to authenticated;
revoke all on function public.intel_agent_fail_job(uuid, text, text, int) from public, anon;
grant execute on function public.intel_agent_fail_job(uuid, text, text, int) to authenticated;

-- 3. Validate arguments and build the summary the administrator will read (older tools unchanged; new tools added) ------------------
create or replace function public._intel_chat_prepare(p_org uuid, p_tool text, p_args jsonb) returns jsonb
language plpgsql stable security definer set search_path = pg_catalog, public as $$
declare
  a jsonb := case when jsonb_typeof(coalesce(p_args, '{}'::jsonb)) = 'object' then p_args else '{}'::jsonb end;
  v_ids uuid[]; v_n int; v_have int; v_type text := nullif(a ->> 'type', ''); v_list text; v_val text; s public.intel_settings; v_clean jsonb;
  v_cap int; v_x text; pr public.intel_prospects; v_state text[];
  v_key text; v_kind text; v_title text; v_body text; kbr public.intel_kb; v_set text; v_old text; v_subj text; v_to text; v_mid uuid; v_lead uuid;
  dm public.intel_messages;
begin
  select * into s from public.intel_settings where organization_id = p_org;
  if not found then return jsonb_build_object('ok', false, 'reason', 'not_set_up'); end if;

  if p_tool in ('approve_messages', 'discard_messages') then
    v_ids := public._intel_juuids(a, 'message_ids', 10);
    v_n := coalesce(array_length(v_ids, 1), 0);
    if v_n = 0 then return jsonb_build_object('ok', false, 'reason', 'no_valid_message_ids'); end if;
    v_state := case when p_tool = 'approve_messages' then array['pending_approval'] else array['draft', 'pending_approval', 'approved', 'failed'] end;
    select count(*), string_agg(format('- %s to %s, subject "%s"', m.kind, m.to_email, left(m.subject, 90)), E'\n' order by m.created_at)
      into v_have, v_list
      from public.intel_messages m
     where m.id = any (v_ids) and m.organization_id = p_org and m.direction = 'outbound' and m.state = any (v_state);
    if v_have <> v_n then return jsonb_build_object('ok', false, 'reason', 'some_messages_not_in_a_valid_state'); end if;
    if v_n = 1 and p_tool = 'approve_messages' then
      select E'\nStarts: ' || replace(left(m.body_text, 350), E'\n', ' ') into v_x from public.intel_messages m where m.id = v_ids[1];
    end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('message_ids', to_jsonb(v_ids)),
      'summary', case when p_tool = 'approve_messages' then format('Approve %s message(s) so the worker can send them (daily limits, warm-up and suppression still apply):', v_n)
                      else format('Discard %s message(s) (they will not be sent):', v_n) end || E'\n' || v_list || coalesce(v_x, ''));

  elsif p_tool = 'resume_campaign' then
    if v_type is null or v_type not in ('indie_game_developer', 'animation_studio_support') then return jsonb_build_object('ok', false, 'reason', 'bad_type'); end if;
    if not exists (select 1 from public.intel_campaigns where organization_id = p_org and type = v_type and status <> 'active') then return jsonb_build_object('ok', false, 'reason', 'campaign_already_active'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('type', v_type), 'summary', format('Start the "%s" campaign. Discovery runs again and approved messages for this campaign become sendable.', v_type));

  elsif p_tool = 'set_campaign_cap' then
    if v_type is null or v_type not in ('indie_game_developer', 'animation_studio_support') then return jsonb_build_object('ok', false, 'reason', 'bad_type'); end if;
    if (a ->> 'daily_cap') !~ '^\d{1,2}$' or (a ->> 'daily_cap')::int > 30 then return jsonb_build_object('ok', false, 'reason', 'daily_cap_must_be_0_to_30'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('type', v_type, 'daily_cap', (a ->> 'daily_cap')::int),
      'summary', format('Set the daily introduction limit for "%s" to %s.', v_type, a ->> 'daily_cap'));

  elsif p_tool = 'enable_sending' then
    if not s.kill_switch then return jsonb_build_object('ok', false, 'reason', 'sending_is_already_on'); end if;
    select count(*) into v_n from public.intel_messages where organization_id = p_org and state = 'approved';
    return jsonb_build_object('ok', true, 'args', '{}'::jsonb,
      'summary', format('Turn sending ON. %s message(s) are approved right now and will be emailed within the daily limits (%s introductions, %s follow-ups per day, warm-up %s). Needs a connected mailbox, a registered agent and your sender details.',
                        v_n, s.daily_initial_cap, s.daily_followup_cap, case when s.warmup_enabled then 'on' else 'off' end));

  elsif p_tool = 'set_autonomy' then
    v_x := a ->> 'autonomy';
    if v_x is null or v_x not in ('shadow', 'assisted', 'autonomous') then return jsonb_build_object('ok', false, 'reason', 'autonomy_must_be_shadow_assisted_or_autonomous'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('autonomy', v_x),
      'summary', 'Set autonomy to ' || v_x || ': ' || case v_x
        when 'shadow' then 'the agent drafts everything and every message waits for your approval.'
        when 'assisted' then 'follow-ups and safe auto-replies may go without approval; first introductions still need approval.'
        else 'introductions may be sent WITHOUT your approval (still within limits, suppression and the emergency stop).' end);

  elsif p_tool = 'update_limits' then
    v_clean := '{}'::jsonb;
    if a ? 'daily_initial_cap' then
      if (a ->> 'daily_initial_cap') !~ '^\d{1,2}$' or (a ->> 'daily_initial_cap')::int > 30 then return jsonb_build_object('ok', false, 'reason', 'daily_initial_cap_must_be_0_to_30'); end if;
      v_clean := v_clean || jsonb_build_object('daily_initial_cap', (a ->> 'daily_initial_cap')::int);
    end if;
    if a ? 'followup_business_days' then
      if (a ->> 'followup_business_days') !~ '^\d{1,2}$' or (a ->> 'followup_business_days')::int not between 1 and 30 then return jsonb_build_object('ok', false, 'reason', 'followup_business_days_must_be_1_to_30'); end if;
      v_clean := v_clean || jsonb_build_object('followup_business_days', (a ->> 'followup_business_days')::int);
    end if;
    if a ? 'max_followups' then
      if (a ->> 'max_followups') !~ '^\d$' or (a ->> 'max_followups')::int > 4 then return jsonb_build_object('ok', false, 'reason', 'max_followups_must_be_0_to_4'); end if;
      v_clean := v_clean || jsonb_build_object('max_followups', (a ->> 'max_followups')::int);
    end if;
    if v_clean = '{}'::jsonb then return jsonb_build_object('ok', false, 'reason', 'nothing_to_change'); end if;
    return jsonb_build_object('ok', true, 'args', v_clean, 'summary', 'Change sending limits: ' || (select string_agg(k || ' = ' || val, ', ') from jsonb_each_text(v_clean) as t(k, val)));

  elsif p_tool = 'add_suppression' then
    v_val := lower(btrim(coalesce(a ->> 'value', '')));
    if v_val !~ '^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$' and v_val !~ '^[a-z0-9-]+(\.[a-z0-9-]+)+$' then return jsonb_build_object('ok', false, 'reason', 'value_must_be_an_email_or_a_domain'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('value', v_val, 'reason', left(coalesce(a ->> 'reason', ''), 200)),
      'summary', format('Never contact %s again (%s).', v_val, case when position('@' in v_val) > 0 then 'this address' else 'everyone at this domain' end));

  elsif p_tool = 'review_prospect' then
    v_x := a ->> 'action';
    if v_x is null or v_x not in ('approve', 'reject') then return jsonb_build_object('ok', false, 'reason', 'action_must_be_approve_or_reject'); end if;
    select * into pr from public.intel_prospects where id = public._intel_juuid(a, 'prospect_id') and organization_id = p_org;
    if not found then return jsonb_build_object('ok', false, 'reason', 'prospect_not_found'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('prospect_id', pr.id, 'action', v_x),
      'summary', format('%s prospect "%s" (%s, currently %s).', case v_x when 'approve' then 'Approve' else 'Reject' end, pr.name, pr.type, pr.status));

  elsif p_tool = 'save_knowledge' then
    v_key := lower(btrim(coalesce(a ->> 'key', '')));
    if v_key !~ '^[a-z0-9][a-z0-9_.-]{0,79}$' then return jsonb_build_object('ok', false, 'reason', 'key_must_be_lowercase_letters_digits_dots_dashes_underscores'); end if;
    v_kind := a ->> 'kind';
    if v_kind is null or v_kind not in ('studio', 'service', 'portfolio', 'faq', 'policy', 'template', 'pricing_guidance', 'availability') then return jsonb_build_object('ok', false, 'reason', 'bad_kind'); end if;
    v_title := left(btrim(coalesce(a ->> 'title', '')), 120);
    v_body := btrim(coalesce(a ->> 'body', ''));
    if v_body = '' then return jsonb_build_object('ok', false, 'reason', 'empty_body'); end if;
    if char_length(v_body) > 8000 then return jsonb_build_object('ok', false, 'reason', 'body_over_8000_characters'); end if;
    if v_body ~* '</?[[:space:]]*(approved_knowledge|untrusted)' then return jsonb_build_object('ok', false, 'reason', 'body_contains_reserved_tags'); end if;
    select * into kbr from public.intel_kb where organization_id = p_org and key = v_key;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('key', v_key, 'kind', v_kind, 'title', v_title, 'body', v_body),
      'summary', case when found then format('Replace the knowledge entry "%s" (now version %s) and approve the new text:', v_key, kbr.version)
                      else format('Add a new knowledge entry "%s" (%s) and approve it, so Emma and the email composer can use it:', v_key, v_kind) end
                 || E'\n' || coalesce(nullif(v_title, ''), '(no title)') || E'\n\n' || left(v_body, 1200) || case when char_length(v_body) > 1200 then E'\n...' else '' end);

  elsif p_tool = 'retire_knowledge' then
    v_key := lower(btrim(coalesce(a ->> 'key', '')));
    select * into kbr from public.intel_kb where organization_id = p_org and key = v_key and approved is true;
    if not found then return jsonb_build_object('ok', false, 'reason', 'no_approved_entry_with_that_key'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('key', v_key),
      'summary', format('Retire the knowledge entry "%s" (%s). Emma and the email composer stop using it. The entry stays in Settings > Knowledge base and can be approved again.', v_key, coalesce(nullif(kbr.title, ''), kbr.kind)));

  elsif p_tool = 'change_setting' then
    v_set := a ->> 'setting';
    v_val := btrim(coalesce(a ->> 'value', ''));
    if v_set is null or v_set not in ('sender_name', 'sender_email', 'physical_address', 'opt_out_text', 'timezone', 'send_start_hour', 'send_end_hour',
                                      'send_weekdays_only', 'warmup_enabled', 'allow_cap_shift', 'daily_followup_cap', 'max_auto_replies_per_thread') then
      return jsonb_build_object('ok', false, 'reason', 'setting_not_changeable_from_chat');
    end if;
    if v_set in ('send_weekdays_only', 'warmup_enabled', 'allow_cap_shift') then
      v_val := case when lower(v_val) in ('true', 'yes', 'on', '1') then 'true' when lower(v_val) in ('false', 'no', 'off', '0') then 'false' end;
      if v_val is null then return jsonb_build_object('ok', false, 'reason', 'value_must_be_yes_or_no'); end if;
      v_old := case v_set when 'send_weekdays_only' then s.send_weekdays_only::text when 'warmup_enabled' then s.warmup_enabled::text else s.allow_cap_shift::text end;
    elsif v_set in ('send_start_hour', 'send_end_hour', 'daily_followup_cap', 'max_auto_replies_per_thread') then
      if v_val !~ '^[0-9]{1,3}$' then return jsonb_build_object('ok', false, 'reason', 'value_must_be_a_whole_number'); end if;
      v_n := v_val::int;
      v_val := v_n::text;
      if v_set = 'send_start_hour' and (v_n > 23 or v_n >= s.send_end_hour) then return jsonb_build_object('ok', false, 'reason', 'start_hour_must_be_0_to_23_and_before_the_end_hour'); end if;
      if v_set = 'send_end_hour' and (v_n < 1 or v_n > 24 or v_n <= s.send_start_hour) then return jsonb_build_object('ok', false, 'reason', 'end_hour_must_be_1_to_24_and_after_the_start_hour'); end if;
      if v_set = 'daily_followup_cap' and v_n > 100 then return jsonb_build_object('ok', false, 'reason', 'daily_followup_cap_must_be_0_to_100'); end if;
      if v_set = 'max_auto_replies_per_thread' and v_n > 5 then return jsonb_build_object('ok', false, 'reason', 'max_auto_replies_must_be_0_to_5'); end if;
      v_old := case v_set when 'send_start_hour' then s.send_start_hour::text when 'send_end_hour' then s.send_end_hour::text
                          when 'daily_followup_cap' then s.daily_followup_cap::text else s.max_auto_replies_per_thread::text end;
    else
      if v_set = 'sender_name' and char_length(v_val) not between 1 and 80 then return jsonb_build_object('ok', false, 'reason', 'sender_name_must_be_1_to_80_characters'); end if;
      if v_set = 'sender_email' and (char_length(v_val) > 254 or v_val !~* '^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$') then return jsonb_build_object('ok', false, 'reason', 'sender_email_must_be_a_valid_address'); end if;
      if v_set = 'physical_address' and char_length(v_val) not between 5 and 300 then return jsonb_build_object('ok', false, 'reason', 'physical_address_must_be_5_to_300_characters'); end if;
      if v_set = 'opt_out_text' and char_length(v_val) not between 10 and 500 then return jsonb_build_object('ok', false, 'reason', 'opt_out_text_must_be_10_to_500_characters'); end if;
      if v_set = 'timezone' then
        if char_length(v_val) not between 1 and 64 then return jsonb_build_object('ok', false, 'reason', 'unknown_time_zone'); end if;
        begin
          perform now() at time zone v_val;
        exception when others then
          return jsonb_build_object('ok', false, 'reason', 'unknown_time_zone');
        end;
      end if;
      v_old := case v_set when 'sender_name' then s.sender_name when 'sender_email' then s.sender_email when 'physical_address' then s.physical_address
                          when 'opt_out_text' then s.opt_out_text else s.timezone end;
    end if;
    if v_old is not distinct from v_val then return jsonb_build_object('ok', false, 'reason', 'already_set_to_that_value'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('setting', v_set, 'value', v_val),
      'summary', format('Change %s from "%s" to "%s".', replace(v_set, '_', ' '), left(coalesce(nullif(v_old, ''), '(empty)'), 120), left(v_val, 300))
                 || case when v_set in ('sender_email', 'physical_address', 'opt_out_text') then E'\nThis is part of the sender identity or legal footer of outgoing email, so it is never applied without your Confirm.' else '' end);

  elsif p_tool = 'edit_message' then
    v_mid := public._intel_juuid(a, 'message_id');
    v_subj := nullif(left(btrim(coalesce(a ->> 'subject', '')), 300), '');
    v_body := nullif(btrim(coalesce(a ->> 'body', '')), '');
    if v_mid is null then return jsonb_build_object('ok', false, 'reason', 'message_id_required'); end if;
    if v_subj is null and v_body is null then return jsonb_build_object('ok', false, 'reason', 'nothing_to_change'); end if;
    if v_subj ~ '[\r\n]' then return jsonb_build_object('ok', false, 'reason', 'subject_has_line_breaks'); end if;
    if char_length(coalesce(v_body, '')) > 20000 then return jsonb_build_object('ok', false, 'reason', 'message_too_long'); end if;
    select * into dm from public.intel_messages where id = v_mid and organization_id = p_org and direction = 'outbound' and state in ('draft', 'pending_approval', 'failed');
    if not found then return jsonb_build_object('ok', false, 'reason', 'message_not_editable'); end if;
    return jsonb_build_object('ok', true, 'args', jsonb_strip_nulls(jsonb_build_object('message_id', v_mid, 'subject', v_subj, 'body', v_body)),
      'summary', format('Rewrite the %s draft to %s. It goes back to "waiting for approval"; nothing is sent.', dm.kind, dm.to_email)
                 || coalesce(E'\nNew subject: ' || v_subj, '') || coalesce(E'\nNew text starts: ' || replace(left(v_body, 500), E'\n', ' '), ''));

  elsif p_tool = 'send_email' then
    v_to := lower(btrim(coalesce(a ->> 'to', '')));
    v_subj := btrim(coalesce(a ->> 'subject', ''));
    v_body := btrim(coalesce(a ->> 'body', ''));
    if char_length(v_to) > 254 or v_to !~ '^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$' then return jsonb_build_object('ok', false, 'reason', 'recipient_must_be_one_valid_email_address'); end if;
    if v_subj = '' then return jsonb_build_object('ok', false, 'reason', 'subject_required'); end if;
    if char_length(v_subj) > 200 or v_subj ~ '[\r\n]' then return jsonb_build_object('ok', false, 'reason', 'subject_too_long_or_has_line_breaks'); end if;
    if v_body = '' then return jsonb_build_object('ok', false, 'reason', 'message_required'); end if;
    if char_length(v_body) > 20000 then return jsonb_build_object('ok', false, 'reason', 'message_too_long'); end if;
    if exists (select 1 from public.intel_suppressions x where x.organization_id = p_org
                  and ((x.kind = 'email' and x.value = v_to) or (x.kind = 'domain' and x.value = split_part(v_to, '@', 2)))) then
      return jsonb_build_object('ok', false, 'reason', 'recipient_is_on_the_do_not_contact_list');
    end if;
    return jsonb_build_object('ok', true, 'args', jsonb_build_object('to', v_to, 'subject', v_subj, 'body', v_body),
      'summary', format('Send this email to %s from your studio mailbox:', v_to) || E'\nSubject: ' || v_subj || E'\n\n' || left(v_body, 1500) || case when char_length(v_body) > 1500 then E'\n...' else '' end);

  end if;
  return jsonb_build_object('ok', false, 'reason', 'unknown_tool');
end $$;

-- 3b. Execute a stored DATABASE action. Runs ONLY from intel_admin_chat_confirm, in the confirming administrator's session ----------
create or replace function public._intel_chat_exec(p_org uuid, p_tool text, p_args jsonb) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_prepared jsonb; a jsonb; v_id uuid; v_n int := 0;
begin
  v_prepared := public._intel_chat_prepare(p_org, p_tool, p_args);       -- re-validate against the CURRENT state; never trust stored arguments
  if not coalesce((v_prepared ->> 'ok')::boolean, false) then raise exception 'no longer valid: %', v_prepared ->> 'reason'; end if;
  a := v_prepared -> 'args';
  if p_tool in ('approve_messages', 'discard_messages') then
    for v_id in select (x)::uuid from jsonb_array_elements_text(a -> 'message_ids') x loop
      perform public.intel_admin_review_message(v_id, case when p_tool = 'approve_messages' then 'approve' else 'discard' end);
      v_n := v_n + 1;
    end loop;
    return jsonb_build_object('done', v_n);
  elsif p_tool = 'resume_campaign' then perform public.intel_admin_update_campaign(a ->> 'type', jsonb_build_object('status', 'active'));
  elsif p_tool = 'set_campaign_cap' then perform public.intel_admin_update_campaign(a ->> 'type', jsonb_build_object('daily_cap', (a ->> 'daily_cap')::int));
  elsif p_tool = 'enable_sending' then perform public.intel_admin_set_kill_switch(false);
  elsif p_tool = 'set_autonomy' then perform public.intel_admin_update_settings(jsonb_build_object('autonomy', a ->> 'autonomy'));
  elsif p_tool = 'update_limits' then perform public.intel_admin_update_settings(a);
  elsif p_tool = 'add_suppression' then perform public.intel_admin_add_suppression(a ->> 'value', a ->> 'reason');
  elsif p_tool = 'review_prospect' then perform public.intel_admin_review_prospect((a ->> 'prospect_id')::uuid, a ->> 'action');
  elsif p_tool = 'save_knowledge' then
    perform public.intel_admin_upsert_kb(a ->> 'key', a ->> 'kind', a ->> 'title', a ->> 'body');
    perform public.intel_admin_approve_kb(a ->> 'key', true);
    return jsonb_build_object('done', 1, 'key', a ->> 'key');
  elsif p_tool = 'retire_knowledge' then perform public.intel_admin_approve_kb(a ->> 'key', false);
  elsif p_tool = 'change_setting' then perform public.intel_admin_update_settings(jsonb_build_object(a ->> 'setting', a ->> 'value'));
  elsif p_tool = 'edit_message' then perform public.intel_admin_review_message((a ->> 'message_id')::uuid, 'edit', a ->> 'subject', a ->> 'body');
  elsif p_tool = 'send_email' then raise exception 'send_email runs in the browser, through the email sender, never inside the database';
  else raise exception 'unknown tool';
  end if;
  return jsonb_build_object('done', 1);
end $$;

-- 4. The agent can only PROPOSE ---------------------------------------------------------------------------------------------------
create or replace function public.intel_agent_chat_propose(p_thread_id uuid, p_message_id bigint, p_tool text, p_args jsonb default '{}'::jsonb) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_agent_org(); t public.intel_chat_threads; v_prep jsonb; v_id uuid; v_n int; v_cat text; v_auto boolean;
begin
  select * into t from public.intel_chat_threads where id = p_thread_id and organization_id = v_org;
  if not found then return public._intel_deny('thread_not_found'); end if;
  if not exists (select 1 from public.intel_chat_messages m where m.id = p_message_id and m.thread_id = t.id and m.role = 'user') then return public._intel_deny('message_not_found'); end if;
  if not public.intel_is_authorized_user(t.user_id, v_org) then
    perform public._intel_log(v_org, 'agent', 'chat_denied_unauthorized', 'chat', t.id::text, jsonb_build_object('tool', left(coalesce(p_tool, ''), 40)));
    return public._intel_deny('requester_not_authorized');
  end if;
  if p_tool is null or p_tool not in ('approve_messages','discard_messages','resume_campaign','set_campaign_cap','enable_sending','set_autonomy','update_limits','add_suppression','review_prospect',
                                      'save_knowledge','retire_knowledge','change_setting','edit_message','send_email') then
    perform public._intel_log(v_org, 'agent', 'chat_denied_unknown_tool', 'chat', t.id::text, jsonb_build_object('tool', left(coalesce(p_tool, ''), 40)));
    return public._intel_deny('unknown_tool');
  end if;
  select count(*) into v_n from public.intel_chat_actions where message_id = p_message_id and thread_id = t.id;
  if v_n >= 10 then return public._intel_deny('too_many_proposals_for_one_request'); end if;
  select count(*) into v_n from public.intel_chat_actions where thread_id = t.id and status = 'pending' and expires_at > now();
  if v_n >= 10 then return public._intel_deny('too_many_pending_actions'); end if;
  v_prep := public._intel_chat_prepare(v_org, p_tool, p_args);
  if not coalesce((v_prep ->> 'ok')::boolean, false) then return public._intel_deny(coalesce(v_prep ->> 'reason', 'invalid')); end if;
  v_cat := public._intel_chat_category(p_tool, v_prep -> 'args');
  v_auto := public._intel_chat_may_auto(v_org, v_cat);
  insert into public.intel_chat_actions(thread_id, organization_id, user_id, message_id, tool, args, summary, category)
  values (t.id, v_org, t.user_id, p_message_id, p_tool, v_prep -> 'args', left(v_prep ->> 'summary', 2500), v_cat)
  on conflict (thread_id, message_id, tool, args_hash) do nothing returning id into v_id;
  if v_id is null then
    select id into v_id from public.intel_chat_actions where thread_id = t.id and message_id = p_message_id and tool = p_tool and args_hash = md5((v_prep -> 'args')::text);
  end if;
  insert into public.intel_chat_messages(thread_id, organization_id, role, content, tool_name, tool_args, tool_result, reply_to)
  values (t.id, v_org, 'tool', '', p_tool, v_prep -> 'args', jsonb_build_object('proposed', true), p_message_id);
  perform public._intel_log(v_org, 'agent', 'chat_proposed', 'chat_action', v_id::text, jsonb_build_object('tool', p_tool, 'category', v_cat, 'args', v_prep -> 'args', 'requested_by', t.user_id));
  return jsonb_build_object('ok', true, 'action_id', v_id, 'status', 'pending', 'summary', v_prep ->> 'summary', 'will_run_automatically', v_auto,
    'note', case when v_auto
                 then 'Queued. The administrator allowed this kind of action without asking, so it runs by itself as soon as their open chat window picks it up. It has NOT run yet: the card shows the result.'
                 else 'NOTHING has changed yet. The administrator must press Confirm on the card in the chat.' end);
end $$;

-- 5. The administrator decides (database actions) ---------------------------------------------------------------------------------
drop function if exists public.intel_admin_chat_confirm(uuid);
create or replace function public.intel_admin_chat_confirm(p_action_id uuid, p_auto boolean default false) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_admin_org(); x public.intel_chat_actions; r jsonb;
begin
  select * into x from public.intel_chat_actions where id = p_action_id and organization_id = v_org and user_id = auth.uid() for update;
  if not found then raise exception 'action not found'; end if;
  if x.status <> 'pending' then return jsonb_build_object('ok', x.status = 'confirmed', 'status', x.status, 'result', x.result); end if;   -- double clicks do nothing
  if now() > x.expires_at then
    update public.intel_chat_actions set status = 'expired', decided_at = now() where id = x.id;
    return jsonb_build_object('ok', false, 'status', 'expired');
  end if;
  if public._intel_chat_executor(x.tool) <> 'database' then raise exception 'this action runs in the browser: use intel_admin_chat_claim'; end if;
  if coalesce(p_auto, false) and not public._intel_chat_may_auto(v_org, public._intel_chat_category(x.tool, x.args)) then
    return jsonb_build_object('ok', false, 'status', 'pending', 'error', 'auto_not_allowed');      -- the card stays pending for a manual Confirm
  end if;
  begin
    r := public._intel_chat_exec(v_org, x.tool, x.args);
    update public.intel_chat_actions set status = 'confirmed', result = r, decided_at = now(), auto = coalesce(p_auto, false) where id = x.id;
    perform public._intel_log(v_org, 'admin', case when coalesce(p_auto, false) then 'chat_action_auto_confirmed' else 'chat_action_confirmed' end, 'chat_action', x.id::text,
                              jsonb_build_object('tool', x.tool, 'category', x.category, 'args', x.args));
    return jsonb_build_object('ok', true, 'status', 'confirmed', 'result', r);
  exception when others then
    update public.intel_chat_actions set status = 'failed', result = jsonb_build_object('error', left(sqlerrm, 300)), decided_at = now(), auto = coalesce(p_auto, false) where id = x.id;
    return jsonb_build_object('ok', false, 'status', 'failed', 'error', left(sqlerrm, 300));
  end;
end $$;

-- 5b. Browser actions (send_email): claim once, send in the browser, then report ---------------------------------------------------
create or replace function public.intel_admin_chat_claim(p_action_id uuid, p_auto boolean default false) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_admin_org(); x public.intel_chat_actions; v_prep jsonb;
begin
  select * into x from public.intel_chat_actions where id = p_action_id and organization_id = v_org and user_id = auth.uid() for update;
  if not found then raise exception 'action not found'; end if;
  if x.status <> 'pending' then return jsonb_build_object('ok', false, 'status', x.status); end if;                    -- one claim, ever: no second send
  if now() > x.expires_at then
    update public.intel_chat_actions set status = 'expired', decided_at = now() where id = x.id;
    return jsonb_build_object('ok', false, 'status', 'expired');
  end if;
  if public._intel_chat_executor(x.tool) <> 'browser' then raise exception 'this action runs in the database: use intel_admin_chat_confirm'; end if;
  if coalesce(p_auto, false) and not public._intel_chat_may_auto(v_org, public._intel_chat_category(x.tool, x.args)) then
    return jsonb_build_object('ok', false, 'status', 'pending', 'error', 'auto_not_allowed');
  end if;
  v_prep := public._intel_chat_prepare(v_org, x.tool, x.args);                                                          -- re-validated against the CURRENT state (for example a new do-not-contact entry)
  if not coalesce((v_prep ->> 'ok')::boolean, false) then
    update public.intel_chat_actions set status = 'failed', result = jsonb_build_object('error', 'no longer valid: ' || coalesce(v_prep ->> 'reason', 'invalid')), decided_at = now() where id = x.id;
    return jsonb_build_object('ok', false, 'status', 'failed', 'error', 'no longer valid: ' || coalesce(v_prep ->> 'reason', 'invalid'));
  end if;
  update public.intel_chat_actions set status = 'running', auto = coalesce(p_auto, false), decided_at = now() where id = x.id;
  perform public._intel_log(v_org, 'admin', case when coalesce(p_auto, false) then 'chat_action_auto_claimed' else 'chat_action_claimed' end, 'chat_action', x.id::text,
                            jsonb_build_object('tool', x.tool, 'category', x.category, 'to', v_prep -> 'args' ->> 'to'));
  return jsonb_build_object('ok', true, 'status', 'running', 'tool', x.tool, 'args', v_prep -> 'args');
end $$;

create or replace function public.intel_admin_chat_finish(p_action_id uuid, p_ok boolean, p_result jsonb default '{}'::jsonb) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_admin_org(); x public.intel_chat_actions; r jsonb := case when jsonb_typeof(coalesce(p_result, '{}'::jsonb)) = 'object' then coalesce(p_result, '{}'::jsonb) else '{}'::jsonb end; v_clean jsonb;
begin
  select * into x from public.intel_chat_actions where id = p_action_id and organization_id = v_org and user_id = auth.uid() for update;
  if not found then raise exception 'action not found'; end if;
  if x.status <> 'running' then return jsonb_build_object('ok', false, 'status', x.status); end if;
  v_clean := jsonb_strip_nulls(jsonb_build_object('message_id', left(r ->> 'message_id', 80), 'outcome', left(r ->> 'outcome', 40),
                                                  'edited', (r ->> 'edited') = 'true', 'error', left(r ->> 'error', 300), 'note', left(r ->> 'note', 300)));
  update public.intel_chat_actions set status = case when coalesce(p_ok, false) then 'confirmed' else 'failed' end, result = v_clean, decided_at = now() where id = x.id;
  perform public._intel_log(v_org, 'admin', case when coalesce(p_ok, false) then 'chat_action_done' else 'chat_action_failed' end, 'chat_action', x.id::text,
                            jsonb_build_object('tool', x.tool, 'category', x.category, 'result', v_clean));
  return jsonb_build_object('ok', true, 'status', case when coalesce(p_ok, false) then 'confirmed' else 'failed' end);
end $$;

-- 6. Emma's permissions: which kinds of action she may run without asking -----------------------------------------------------------
create or replace function public.intel_admin_set_emma_authority(p_patch jsonb) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare v_org uuid := public.intel_admin_org(); k text; v text; v_cur jsonb;
begin
  if jsonb_typeof(coalesce(p_patch, 'null'::jsonb)) <> 'object' then raise exception 'patch must be an object'; end if;
  select emma_authority into v_cur from public.intel_settings where organization_id = v_org for update;
  if not found then raise exception 'Run setup first (intel_admin_bootstrap)'; end if;
  for k, v in select key, value #>> '{}' from jsonb_each(p_patch) loop
    if k not in ('knowledge', 'settings', 'messages', 'prospects', 'email') then raise exception 'unknown category: %', left(k, 30); end if;
    if v not in ('ask', 'auto') then raise exception 'value must be ask or auto'; end if;
    v_cur := coalesce(v_cur, '{}'::jsonb) || jsonb_build_object(k, v);
  end loop;
  update public.intel_settings set emma_authority = v_cur, updated_at = now(), updated_by = auth.uid() where organization_id = v_org;
  perform public._intel_log(v_org, 'admin', 'set_emma_authority', 'settings', v_org::text, p_patch);
  return v_cur;
end $$;

-- 7. Read tools for Emma (no changes, audited, capped) --------------------------------------------------------------------------------
create or replace function public.intel_agent_chat_read(p_thread_id uuid, p_message_id bigint, p_tool text, p_args jsonb default '{}'::jsonb) returns jsonb
language plpgsql security definer set search_path = pg_catalog, public as $$
declare
  v_org uuid := public.intel_agent_org(); t public.intel_chat_threads; s public.intel_settings;
  a jsonb := case when jsonb_typeof(coalesce(p_args, '{}'::jsonb)) = 'object' then coalesce(p_args, '{}'::jsonb) else '{}'::jsonb end;
  v_out jsonb; v_n int; v_key text; v_email text; v_limit int := public._intel_jint(a, 'limit', 10, 1, 25); kbr public.intel_kb;
begin
  select * into t from public.intel_chat_threads where id = p_thread_id and organization_id = v_org;
  if not found then return public._intel_deny('thread_not_found'); end if;
  if not exists (select 1 from public.intel_chat_messages m where m.id = p_message_id and m.thread_id = t.id and m.role = 'user') then return public._intel_deny('message_not_found'); end if;
  if not public.intel_is_authorized_user(t.user_id, v_org) then
    perform public._intel_log(v_org, 'agent', 'chat_denied_unauthorized', 'chat', t.id::text, jsonb_build_object('tool', left(coalesce(p_tool, ''), 40)));
    return public._intel_deny('requester_not_authorized');
  end if;
  if p_tool is null or p_tool not in ('settings_overview', 'knowledge_list', 'knowledge_get', 'list_drafts', 'check_recipient') then
    perform public._intel_log(v_org, 'agent', 'chat_denied_unknown_tool', 'chat', t.id::text, jsonb_build_object('tool', left(coalesce(p_tool, ''), 40)));
    return public._intel_deny('unknown_tool');
  end if;
  select count(*) into v_n from public.intel_chat_messages m where m.thread_id = t.id and m.role = 'tool' and m.reply_to = p_message_id;
  if v_n >= 12 then return public._intel_deny('too_many_tool_calls'); end if;
  select * into s from public.intel_settings where organization_id = v_org;

  begin
    if p_tool = 'settings_overview' then
      v_out := jsonb_build_object('ok', true, 'sending_stopped', s.kill_switch, 'autonomy', s.autonomy,
        'settings', jsonb_build_object('sender_name', s.sender_name, 'sender_email', s.sender_email, 'physical_address_set', s.physical_address <> '', 'timezone', s.timezone,
          'send_start_hour', s.send_start_hour, 'send_end_hour', s.send_end_hour, 'send_weekdays_only', s.send_weekdays_only, 'warmup_enabled', s.warmup_enabled,
          'allow_cap_shift', s.allow_cap_shift, 'daily_initial_cap', s.daily_initial_cap, 'daily_followup_cap', s.daily_followup_cap,
          'followup_business_days', s.followup_business_days, 'max_followups', s.max_followups, 'max_auto_replies_per_thread', s.max_auto_replies_per_thread),
        'emma_runs_without_asking', coalesce(s.emma_authority, '{}'::jsonb));

    elsif p_tool = 'knowledge_list' then
      v_out := jsonb_build_object('ok', true, 'entries', coalesce((select jsonb_agg(jsonb_build_object('key', k.key, 'kind', k.kind, 'title', k.title, 'approved', k.approved,
                 'version', k.version, 'characters', char_length(k.body), 'updated_at', k.updated_at) order by k.key)
                 from (select * from public.intel_kb where organization_id = v_org order by key limit 80) k), '[]'::jsonb));

    elsif p_tool = 'knowledge_get' then
      v_key := lower(btrim(coalesce(a ->> 'key', '')));
      select * into kbr from public.intel_kb where organization_id = v_org and key = v_key;
      if not found then return public._intel_deny('no_entry_with_that_key'); end if;
      v_out := jsonb_build_object('ok', true, 'entry', jsonb_build_object('key', kbr.key, 'kind', kbr.kind, 'title', kbr.title, 'body', kbr.body, 'approved', kbr.approved, 'version', kbr.version));

    elsif p_tool = 'list_drafts' then
      v_out := jsonb_build_object('ok', true, 'drafts', coalesce((select jsonb_agg(jsonb_build_object('message_id', d.id, 'kind', d.kind, 'state', d.state, 'to_email', d.to_email,
                 'subject', left(d.subject, 160), 'preview', left(replace(d.body_text, E'\n', ' '), 300), 'prospect', d.prospect_name, 'created_at', d.created_at) order by d.created_at desc)
                 from (select m.*, (select p.name from public.intel_prospects p where p.id = m.prospect_id) as prospect_name
                         from public.intel_messages m
                        where m.organization_id = v_org and m.direction = 'outbound' and m.state in ('draft', 'pending_approval', 'failed')
                        order by m.created_at desc limit v_limit) d), '[]'::jsonb));

    elsif p_tool = 'check_recipient' then
      v_email := lower(btrim(coalesce(a ->> 'email', '')));
      if v_email !~ '^[a-z0-9._%+-]+@[a-z0-9.-]+\.[a-z]{2,}$' then return public._intel_deny('not_an_email_address'); end if;
      v_out := jsonb_build_object('ok', true, 'email', v_email,
        'do_not_contact', exists (select 1 from public.intel_suppressions x where x.organization_id = v_org
                                    and ((x.kind = 'email' and x.value = v_email) or (x.kind = 'domain' and x.value = split_part(v_email, '@', 2)))),
        'known_prospects', coalesce((select jsonb_agg(jsonb_build_object('name', p.name, 'status', p.status, 'type', p.type)) from (select * from public.intel_prospects
                                       where organization_id = v_org and lower(contact_email) = v_email limit 3) p), '[]'::jsonb),
        'messages_we_sent', (select count(*) from public.intel_messages where organization_id = v_org and direction = 'outbound' and state = 'sent' and lower(to_email) = v_email),
        'last_sent_at', (select max(sent_at) from public.intel_messages where organization_id = v_org and direction = 'outbound' and state = 'sent' and lower(to_email) = v_email));
    end if;
  exception when others then
    return public._intel_deny('tool_failed: ' || left(sqlerrm, 160));
  end;

  insert into public.intel_chat_messages(thread_id, organization_id, role, content, tool_name, tool_args, tool_result, reply_to)
  values (t.id, v_org, 'tool', '', p_tool, a, v_out, p_message_id);
  perform public._intel_log(v_org, 'agent', 'chat_read', 'chat', t.id::text, jsonb_build_object('tool', p_tool, 'message_id', p_message_id, 'requested_by', t.user_id));
  return v_out;
end $$;

do $$
declare r record;
begin
  for r in select p.oid::regprocedure as sig, p.proname
             from pg_proc p join pg_namespace n on n.oid = p.pronamespace
            where n.nspname = 'public' and (p.proname like 'intel\_%' or p.proname like '\_intel\_%' or p.proname = 'crm_find_lead_duplicates')
  loop
    execute format('revoke all on function %s from public, anon', r.sig);
    if r.proname like '\_intel\_%' or r.proname = 'intel_is_authorized_user' then
      execute format('revoke all on function %s from authenticated', r.sig);
    else
      execute format('grant execute on function %s to authenticated', r.sig);
    end if;
  end loop;
end $$;
